class Solution {
    static int solve(int n ,int[]dp){
        if(n==0)return 1;
        if(n <0)return 0;
        return dp[n]= solve(n-1,dp)+solve(n-2,dp);
    }
    public int climbStairs(int n) {
        return solve(n,dp );
