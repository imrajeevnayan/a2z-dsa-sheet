class Solution {
    public int amount(int[] nums,int i,int[]dp){
        int take=nums[i]+amount(nums,i+2,dp);
        int skip=0+amount(nums,i+1,dp);
        return dp[i]= Math.max(take,skip);
    }
    public int rob(int[] nums) {
        return amount(nums,0,dp);
