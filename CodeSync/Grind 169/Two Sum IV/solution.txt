import java.util.HashSet;
import java.util.Set;
 
class Solution {
    public static boolean twoSumInBST(BinaryTreeNode root, int target) {
        return helper(root, target, new HashSet<>());
    }
 
    private static boolean helper(BinaryTreeNode root, int target, Set<Integer> set) {
        if (root == null) return false;
 
        int val = (int) root.data;  // Cast Object to int
 
        if (set.contains(target - val)) return true;
        set.add(val);
 
        return helper(root.left, target, set) || helper(root.right, target, set);
    }
}
