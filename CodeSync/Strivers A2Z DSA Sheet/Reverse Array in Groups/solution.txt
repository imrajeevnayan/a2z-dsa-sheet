class Solution {
    public void reverseInGroups(int[] arr, int k) {

        int n = arr.length;

        // Har k-size ke group par iterate karo
        for (int i = 0; i < n; i += k) {

            // Current group ka start index
            int left = i;

            // Current group ka end index
            // Agar last group me k se kam elements hain,
            // to n-1 tak hi reverse karenge
            int right = Math.min(i + k - 1, n - 1);

            // Two pointers se current group reverse karo
            while (left < right) {

                // Swap arr[left] aur arr[right]
                int temp = arr[left];
                arr[left] = arr[right];
                arr[right] = temp;

                // Pointers move karo
                left++;
                right--;
            }
        }
    }
}
