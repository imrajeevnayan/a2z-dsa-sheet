import java.util.Scanner;
 
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        // Input N
        int n = sc.nextInt();
        int[] arr = new int[n];
        
        // Array input lo
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        
        // Recursive function call karo (start=0, end=n-1)
        reverseArray(arr, 0, n - 1);
        
        // Reversed array print karo
        for (int i = 0; i < n; i++) {
            System.out.print(arr[i] + " ");
        }
    }
 
    // Recursive function to reverse array in-place
    public static void reverseArray(int[] arr, int s, int e) {
        // Base case: Agar pointers cross ho jayein ya mil jayein, toh ruk jao
        if (s >= e) {
            return;
        }
        
        // Swap start and end elements
        int temp = arr[s];
        arr[s] = arr[e];
        arr[e] = temp;
        
        // Recursively call for remaining middle part
        reverseArray(arr, s + 1, e - 1);
    }
