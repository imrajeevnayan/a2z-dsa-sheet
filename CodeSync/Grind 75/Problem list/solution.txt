import java.util.* ;
import java.io.*; 
/***********************************************************
 
  import java.util.HashSet;
import java.util.Set;
 
class Solution {
    public static boolean twoSumInBST(BinaryTreeNode root, int K) {
        return dfs(root, K, new HashSet<>());
    }
 
    private static boolean dfs(BinaryTreeNode node, int K, Set<Integer> seen) {
        if (node == null) return false;
 
        int val = (int) node.data; // cast Object to int
        if (seen.contains(K - val)) return true;
 
        seen.add(val);
 
        return dfs(node.left, K, seen) || dfs(node.right, K, seen);
    }
}

class Solution {
 
    public static boolean kSumPair(TreeNode < Integer > root, int k) {
        return dfs(root,k,new HashSet<>());
    }
    private static boolean dfs(TreeNode<Integer>root,int k,Set<Integer>set){
        if(root==null)return false;
        int val=(int)root.data;
        if(set.contains(k-val))return true;
        set.add(val);
        return dfs(root.left,k,set) || dfs(root.right,k,set);
    }
