class Solution {
    public int kthElement(int a[], int b[], int k) {
        int i = 0, j = 0,count = 0;
        while (i < a.length && j < b.length) {
            int m;
            if (a[i] < b[j])  m= a[i++];
            else m = b[j++];
            count++;
            if (count == k) return m;
        }
        while (i < a.length){count++;
            if (count == k) return a[i];
            i++;
        }
        while (j < b.length) {
            count++;
            if (count == k)return b[j];
            j++;
        }
        return -1;
    }
}
