class Solution {
    public ArrayList<Integer> levelOrder(Node root) {
        ArrayList<Integer> ans = new ArrayList<>();
        if (root == null) return ans;
        Queue<Node> q = new LinkedList<>();
        q.add(root);

        while (!q.isEmpty()) {
            Node temp = q.poll();
            ans.add(temp.data);

            if (temp.left != null)  q.add(temp.left);

            if (temp.right != null)
                q.add(temp.right);
        }

        return ans;
    }
}
