/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int countDominantNodes(TreeNode root) {
        return dfs(root)[0];
    }
    
    // Returns an array where:
    // [0] = count of dominant nodes in this subtree
    // [1] = maximum value in this subtree
    private int[] dfs(TreeNode node) {
        if (node == null) {
            return new int[]{0, Integer.MIN_VALUE};
        }
        
        int[] left = dfs(node.left);
        int[] right = dfs(node.right);
        
        // Maximum value in current subtree
        int maxVal = Math.max(node.val, Math.max(left[1], right[1]));
        
        // Count dominant nodes: sum from children + 1 if current node is dominant
        int count = left[0] + right[0];
        if (node.val == maxVal) {
            count++;
        }
        
        return new int[]{count, maxVal};
    }
}
