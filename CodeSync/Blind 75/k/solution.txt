import java.util.* ;
import java.io.*; 
public class Solution {
    public static Node findKthFromLast(Node head, int K) {
       Node slow=head,fast=head;
        for(int i=1;i<=K;i++){
            if(fast==null) return null;
             fast=fast.next;
        }
        while(fast!=null){
            fast=fast.next;
            slow=slow.next;
        }
        return slow;
    }
}
