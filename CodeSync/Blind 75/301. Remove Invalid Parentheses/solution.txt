class Solution {
    private boolean isValid(String s) {
        int count = 0;
        for (char c : s.toCharArray()) {
            if (c == '(') {
                count++;
            } else if (c == ')') {
                count--;
            }
            // If count goes negative, it means extra ')' appeared
            if (count < 0) return false;
        }
        // At the end, count should be 0 for balanced parentheses
        return count == 0;
    }
