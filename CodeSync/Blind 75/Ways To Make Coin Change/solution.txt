public class Solution {
    public static long countWaysToMakeChange(int denominations[], int value) {
        // dp[i] will store the number of ways to make value 'i'
        long[] dp = new long[value + 1];
        
        // Base case: 0 value banane ka 1 tareeka hai (koi coin na lena)
        dp[0] = 1;
        
        // Har denomination ke liye dp array update karo
        for (int coin : denominations) {
            for (int i = coin; i <= value; i++) {
                dp[i] += dp[i - coin];
            }
        }
        
        return dp[value];
    }
}
