import java.util.ArrayList;
 
public class Solution {
 
    public static ArrayList<Integer> sumTree(BinaryTreeNode<Integer> root) {
        ArrayList<Integer> result = new ArrayList<>();
 
        convert(root);
        preorder(root, result);
 
        return result;
    }
 
    private static void convert(BinaryTreeNode<Integer> root) {
        if (root == null) {
            return;
        }
 
        int left = root.left == null ? 0 : root.left.data;
        int right = root.right == null ? 0 : root.right.data;
 
        convert(root.left);
        convert(root.right);
 
        root.data = left + right;
    }
 
    private static void preorder(BinaryTreeNode<Integer> root,
                                  ArrayList<Integer> result) {
        if (root == null) {
            return;
        }
 
        result.add(root.data);
        preorder(root.left, result);
        preorder(root.right, result);
    }
}
