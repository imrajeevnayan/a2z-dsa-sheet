import java.util.*;
 
public class Solution {
 
    public static boolean pairSumBst(TreeNode root, int k) {
        Set<Integer> set = new HashSet<>();
        return solve(root, k, set);
    }
 
    private static boolean solve(TreeNode root,int k,Set<Integer> set) {
        if (root == null)return false;
        int rem = k - root.data;
 
        if (set.contains(rem)) return true;
        set.add(root.data);
 
        return solve(root.left, k, set) || solve(root.right, k, set);
    }
}
