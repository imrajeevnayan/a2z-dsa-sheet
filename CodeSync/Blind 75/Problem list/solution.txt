import java.util.*;
 
public class Solution {
    public static List<Integer> traverseBoundary(TreeNode root) {
        List<Integer> ans = new ArrayList<>();
 
        if (root == null) return ans;
 
        // Root
        ans.add(root.data);
 
        // Left boundary (excluding leaf nodes)
        addLeftBoundary(root.left, ans);
 
        // Leaf nodes
        addLeaves(root.left, ans);
        addLeaves(root.right, ans);
 
        // Right boundary (excluding leaf nodes), in reverse
        addRightBoundary(root.right, ans);
 
        return ans;
    }
 
    private static void addLeftBoundary(TreeNode node, List<Integer> ans) {
        while (node != null) {

                ans.add(node.data);
            }
 
            if (node.left != null) {
                node = node.left;
            } else {
                node = node.right;
            }
        }
