import java.util.*;
import java.io.*;
 
public class Solution {
    public static String encode(String message) {
        // Agar string empty hai toh wapas aa jao
        if (message.length() == 0) {
            return "";
        }
 
        StringBuilder result = new StringBuilder();
        int count = 1;
        
        // Loop 0 se n-2 tak chalega taaki hum current aur next ko compare kar sakein
        for (int i = 0; i < message.length() - 1; i++) {
            char curr = message.charAt(i);
            char next = message.charAt(i + 1);
            
            if (curr == next) {
                count++;
            } else {
                // Character badal gaya, toh purane wale ko add karo
                result.append(curr);
                result.append(count);
                count = 1; // Count reset karo
            }
        }
        
        // Last character ko alag se handle karo (kyunki loop usse chhod deta hai)
        result.append(message.charAt(message.length() - 1));
        result.append(count);
        
        return result.toString();
    }
}
