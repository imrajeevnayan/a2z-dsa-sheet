import java.util.*;
 
public class Solution {
    
    public static List<List<Integer>> findTriplets(TreeNode<Integer> root, int x) {
        
        List<List<Integer>> ans = new ArrayList<>();
        List<Integer> values = new ArrayList<>();
        
        // Tree ke saare elements list mein store karo
        storeValues(root, values);
        
        // Sorting se 3-sum efficiently kar sakte hain
        Collections.sort(values);
        
        int n = values.size();
        
        for (int i = 0; i < n - 2; i++) {
            
            // Duplicate triplets avoid karo
            if (i > 0 && values.get(i).equals(values.get(i - 1))) {
                continue;
            }
            
            int left = i + 1;
            int right = n - 1;
            
            while (left < right) {
                
                long sum = (long) values.get(i)
                         + values.get(left)
                         + values.get(right);
                
                if (sum == x) {
                    
                    ans.add(Arrays.asList(
                        values.get(i),
                        values.get(left),
                        values.get(right)
                    ));
                    
                    left++;
