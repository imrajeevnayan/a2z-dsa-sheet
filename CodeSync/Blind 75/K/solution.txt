import java.util.ArrayList;
import java.util.PriorityQueue;
 
public class Solution {
 
    public static int getKthLargest(ArrayList<Integer> arr, int k) {
 
        int n = arr.size();
 
        PriorityQueue<Integer> pq = new PriorityQueue<>();
 
        for (int i = 0; i < n; i++) {
 
            int sum = 0;
 
            for (int j = i; j < n; j++) {
 
                sum += arr.get(j);
 
                if (pq.size() < k) {
                    pq.offer(sum);
                } 
                else if (sum > pq.peek()) {
                    pq.poll();
                    pq.offer(sum);
                }
            }
        }
 
        return pq.peek();
    }
}
