import java.util.*;
import java.io.*;
 
public class Solution 
{
    public static TreeNode<Integer> invertBinaryTree(
            TreeNode<Integer> root, TreeNode<Integer> leaf) {
 
        if (root == null) {
            return null;
        }
 
        if (root == leaf) {
            return root;
        }
 
        TreeNode<Integer> leftResult = invertBinaryTree(root.left, leaf);
 
        if (leftResult != null) {
            root.left = null;
            root.right = null;
            leaf.left = root;
            return leftResult;
        }
 
        TreeNode<Integer> rightResult = invertBinaryTree(root.right, leaf);
 
        if (rightResult != null) {
            root.left = null;
            root.right = null;
            leaf.right = root;
            return rightResult;
        }
 
        return null;
    }
}
