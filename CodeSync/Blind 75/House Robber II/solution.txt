import java.util.* ;
import java.io.*; 
 
public class Solution {
    public static long houseRobber(int[] valueInHouse) {
        
        int n = valueInHouse.length;
 
        if (n == 0) {
            return 0;
        }
 
        if (n == 1) {
            return valueInHouse[0];
        }
 
        long prev2 = 0;
        long prev1 = valueInHouse[0];
 
        for (int i = 1; i < n; i++) {
            
            long rob = valueInHouse[i] + prev2;
            long skip = prev1;
 
            long curr = Math.max(rob, skip);
 
            prev2 = prev1;
            prev1 = curr;
        }
 
        return prev1;
    }
}
