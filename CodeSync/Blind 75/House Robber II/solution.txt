    private static long rob(int[] arr, int start, int end) {
 
        long prev1 = 0; // dp[i-1]
        long prev2 = 0; // dp[i-2]
 
        for (int i = start; i <= end; i++) {
 
            // Current house loot karo
            long take = arr[i] + prev2;
 
            // Current house skip karo
            long skip = prev1;
 
            long curr = Math.max(take, skip);
 
            prev2 = prev1;
            prev1 = curr;
        }
 
        return prev1;
    }
