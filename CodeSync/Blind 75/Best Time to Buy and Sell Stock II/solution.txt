public class Solution {
    public static long getMaximumProfit(int n, long[] values) {
        // Profit track karne ke liye variable
        long profit = 0;
        
        // Index 1 se start karo kyunki hum current ko previous se compare karenge
        for (int i = 1; i < n; i++) {
            // Agar aaj ka price kal se zyada hai ? profit lo
            if (values[i] > values[i - 1]) {
                profit += values[i] - values[i - 1];
            }
        }
        
        return profit;
    }
}
