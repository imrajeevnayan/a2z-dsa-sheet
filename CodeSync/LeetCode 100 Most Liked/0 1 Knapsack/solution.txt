import java.util.*;
public class Solution {
    public static String read(int n, int []book, int target){
        Map<Integer,Integer>map=new HashMap<>();
        for(int i=0;i<n;i++){
            int rem=target-book[i];
            if(map.containsKey(rem))return "YES";
            map.put(book[i],i);
        }
        return "NO";
    }
}
