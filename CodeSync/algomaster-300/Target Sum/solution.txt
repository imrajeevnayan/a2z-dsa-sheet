import java.util.*;
 
public class Solution {
 
    public static int targetSum(int n, int target, int[] arr) {
 
        int totalSum = 0;
 
        for (int x : arr) {
            totalSum += x;
        }
 
        if (Math.abs(target) > totalSum) {
            return 0;
        }
 
        if ((totalSum + target) % 2 != 0) {
            return 0;
        }
 
        int reqSum = (totalSum + target) / 2;
 
        int[] dp = new int[reqSum + 1];
        dp[0] = 1;
 
        for (int num : arr) {
 
            for (int s = reqSum; s >= num; s--) {
                dp[s] += dp[s - num];
            }
        }
 
        return dp[reqSum];
    }
}
