class Solution {
    public boolean findPattern(String s) {
        int i = 0;

        while (i < s.length()) {
            int x = 0, y = 0;

            while (i < s.length() && s.charAt(i) == 'x') {
                x++;
                i++;
            }

            while (i < s.length() && s.charAt(i) == 'y') {
                y++;
                i++;
            }

            if (x != y || x == 0) {
                return false;
            }
        }

        return true;
    }
}
