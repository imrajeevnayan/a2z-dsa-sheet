import java.util.*;

class Solution {
    public String frequencySort(String s) {
        // Step 1: Frequency count karo HashMap mein
        HashMap<Character, Integer> map = new HashMap<>();
        for (char c : s.toCharArray()) {
            map.put(c, map.getOrDefault(c, 0) + 1);
        }

        // Step 2: Max-Heap banao (Zyada frequency wala upar rahega)
        PriorityQueue<Character> pq = new PriorityQueue<>((a, b) -> {
            int freqDiff = map.get(b) - map.get(a);
            // Agar frequency same ho, toh character ke ASCII value ke basis par sort karo (optional but good practice)
            if (freqDiff != 0) return freqDiff;
            return a - b; 
        });

        pq.addAll(map.keySet());

        // Step 3: String build karo
        StringBuilder sb = new StringBuilder();

        while (!pq.isEmpty()) {
            char c = pq.poll();
            int freq = map.get(c);

            // Character ko 'freq' times append karo
            for (int i = 0; i < freq; i++) {
                sb.append(c);
            }
        }

        return sb.toString();
    }
}
