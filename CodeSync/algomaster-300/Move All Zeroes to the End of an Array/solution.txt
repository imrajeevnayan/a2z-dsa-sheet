class Solution {
    void pushZerosToEnd(int[] arr) {

        int j = 0; // Next non-zero element ki position

        for (int i = 0; i < arr.length; i++) {

            // Agar current element non-zero hai
            if (arr[i] != 0) {

                // Swap arr[i] aur arr[j]
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;

                // Next position par move karo
                j++;
            }
        }
    }
}
