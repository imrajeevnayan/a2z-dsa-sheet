public class Solution {
 
    public static void postOrder(BinaryTreeNode<Integer> root) {
         if (root == null) {
            return;
        }
        
        // Step 1: LEFT - Pehle left subtree traverse karo
        postOrder(root.left);
        
        // Step 2: RIGHT - Phir right subtree traverse karo
        postOrder(root.right);
        
        // Step 3: ROOT - Last mein current node print karo
        System.out.print(root.data + " ");
    }
 
}
