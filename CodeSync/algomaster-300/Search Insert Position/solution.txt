class Solution {
    public int searchInsert(int[] nums, int target) {
        int low=0,high=nums.length-1,idx=nums.length;
        while(low <= high){
            int mid=low+(high-low)/2;
            if(nums[mid]==target) return mid;
            else if(nums[mid]< target)low=mid+1;
            else{
                idx=mid;
                high=mid-1;
            }
        }
       return idx;
    }
