import java.util.*;
 
public class Solution {
    public static List<Integer> searchInSortedArray(int[] arr, int n, int[] queries, int q) {
 
        List<Integer> ans = new ArrayList<>();
 
        for (int i = 0; i < q; i++) {
            int left = 0;
            int right = n - 1;
            int target = queries[i];
            int found = 0;
 
            while (left <= right) {
                int mid = left + (right - left) / 2;
 
                if (arr[mid] == target) {
                    found = 1;
                    break;
                } else if (arr[mid] < target) {
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
 
            ans.add(found);
        }
