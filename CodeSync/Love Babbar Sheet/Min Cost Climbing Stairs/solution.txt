class Solution {
    static int solve(int i, int[] cost) {
        if (i >= cost.length) return 0;
        return cost[i] + Math.min( solve(i + 1, cost), solve(i + 2, cost) );
    }
    public int minCostClimbingStairs(int[] cost) {
        return Math.min( solve(0, cost),solve(1, cost));
    }
}
