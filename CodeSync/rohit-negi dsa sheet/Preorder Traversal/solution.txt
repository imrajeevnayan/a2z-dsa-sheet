class Solution {
    public ArrayList<Integer> preOrder(Node root) {
        ArrayList<Integer> ans = new ArrayList<>();
        if (root == null) return ans;
        ans.add(root.data);
        ans.addAll(preOrder(root.left));
        ans.addAll(preOrder(root.right));
        return ans;
    }
}
