import java.util.* ;
import java.io.*; 
import java.util.ArrayList;
 
public class Solution {
 
    public static ArrayList<Integer> removeDuplicates(ArrayList<Integer> arr) {
 
        HashSet<Integer> set = new HashSet<>();
        ArrayList<Integer> res = new ArrayList<>();
 
        for (int num : arr) {
            if (!set.contains(num)) {
                set.add(num);
                res.add(num);
            }
        }
 
        return res;
    }
}
