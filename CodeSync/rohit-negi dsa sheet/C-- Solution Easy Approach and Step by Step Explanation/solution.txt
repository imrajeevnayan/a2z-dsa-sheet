class Solution {
    public int countSubarray(int[] arr, int l, int r) {
        return count(arr, r) - count(arr, l - 1);
    }

    private int count(int[] arr, int limit) {
        int ans = 0;
        int len = 0;

        for (int x : arr) {
            if (x <= limit) {
                len++;
            } else {
                len = 0;
            }
            ans += len;
        }

        return ans;
    }
}
