2    public int minCostClimbingStairs(int[] cost) {
3        int n = cost.length;
4        int[] dp = new int[n + 1];
5        Arrays.fill(dp, -1);
6        return helper(cost, n, dp);
7    }
8    
9    private int helper(int[] cost, int i, int[] dp) {
10        if (i <= 1) return 0;
11        if (dp[i] != -1) return dp[i];
12        dp[i] = Math.min(
13            helper(cost, i - 1, dp) + cost[i - 1],
14            helper(cost, i - 2, dp) + cost[i - 2]
15        );
16        
17        return dp[i];
18    }
