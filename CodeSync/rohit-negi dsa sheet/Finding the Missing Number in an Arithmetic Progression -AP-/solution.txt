class Solution {
    public int findMissing(int[] arr) {
        int n = arr.length;
        int d = Math.min(arr[1] - arr[0], arr[2] - arr[1]);
        d = Math.min(d, arr[n - 1] - arr[n - 2]);

        int low = 0, high = n - 1;
        while (low <= high) {
            int mid = low + (high - low) / 2;
            int expected = arr[0] + mid * d;
            if (arr[mid] == expected) low = mid + 1;
            else high = mid - 1;
        }
        return arr[0] + low * d;
    }
}
