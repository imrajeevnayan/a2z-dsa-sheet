import java.util.*;
import java.io.*;
 
import java.util.ArrayList;
 
public class Solution {
    public static ArrayList<Integer> getMaximumOfSubarrays( ArrayList<Integer> nums, int k) {
        ArrayList<Integer> ans = new ArrayList<>();
        Deque<Integer> dq = new ArrayDeque<>();
        int left = 0;
 
        for (int right = 0; right < nums.size(); right++) {
 
            // 1. Window ke bahar wale indices remove karo
            while (!dq.isEmpty() && dq.peekFirst() < left) {
                dq.pollFirst();
            }
 
            // 2. Back se smaller elements remove karo
            while (!dq.isEmpty() &&
                   nums.get(dq.peekLast()) <= nums.get(right)) {
                dq.pollLast();
            }
 
            // 3. Current index add karo
            dq.addLast(right);
 
            // 4. Window size == k
            if (right - left + 1 == k) {
 
                // Deque ka front = maximum
                ans.add(nums.get(dq.peekFirst()));
 
                // 5. Window slide
                left++;
            }
        }
 
        return ans;
