class Solution {
    public Node rotate(Node head, int k) {
        if (head == null || head.next == null) {
            return head;
        }
        Node tail = head;
        int n = 1;

        while (tail.next != null) {
            tail = tail.next;
            n++;
        }

        k = k % n;

        if (k == 0) {
            return head;
        }

        tail.next = head;

        for (int i = 0; i < n - k; i++) {
            tail = tail.next;
        }

        head = tail.next;
        tail.next = null;

        return head;
    }
}
