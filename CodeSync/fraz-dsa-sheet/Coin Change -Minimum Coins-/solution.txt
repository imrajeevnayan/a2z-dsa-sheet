class Solution {
    static final int INF = (int)1e9;
    static int solve(int[] coins, int index, int sum, int[][] dp) {
        if (sum == 0) return 0;
        if (index >= coins.length)return INF;
        if (dp[index][sum] != -1)
            return dp[index][sum];
        int take = INF;
        if (coins[index] <= sum) {
            take = 1 + solve(coins, index, sum - coins[index], dp);
        }
        int skip = solve(coins, index + 1, sum, dp);
        return dp[index][sum] = Math.min(take, skip);
    }
    public int minCoins(int coins[], int sum) {
        int[][] dp = new int[coins.length][sum + 1];
        for (int[] row : dp)
        Arrays.fill(row, -1);
        int ans = solve(coins, 0, sum, dp);
        return ans >= INF ? -1 : ans;
    }
}
