public class Solution {
 
    public static boolean checkMember(int n) {
 
        if (n == 0 || n == 1) {
            return true;
        }
 
        long a = 0;
        long b = 1;
 
        while (b <= n) {
            long c = a + b;
 
            if (c == n) {
                return true;
            }
 
            a = b;
            b = c;
        }
 
        return false;
    }
}
