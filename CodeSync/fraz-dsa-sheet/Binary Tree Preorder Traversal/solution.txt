class Solution {
    public List<Integer> preorderTraversal(TreeNode root) {
        List<Integer>res=new ArrayList<>();
        preorder(root,res);
        return res;
    }
    static void preorder(TreeNode node,List<Integer>res){
        if(node==null)return;
    }
        preorder(node.left,res);
        res.add(node.val);
        preorder(node.right,res);
}
