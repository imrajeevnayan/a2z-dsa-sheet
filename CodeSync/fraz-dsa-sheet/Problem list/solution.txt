            // Same first element => duplicate triplets
            if (i > 0 && arr[i] == arr[i - 1]) {
                continue;
            }
 
            HashSet<Integer> set = new HashSet<>();
            HashSet<Integer> used = new HashSet<>();
 
            for (int j = i + 1; j < n; j++) {
 
                int required = K - arr[i] - arr[j];
 
                if (set.contains(required) && !used.contains(required)) {
 
                    ArrayList<Integer> triplet = new ArrayList<>();
 
                    triplet.add(arr[i]);
                    triplet.add(required);
                    triplet.add(arr[j]);
 
                    ans.add(triplet);
 
                    // Don't add same triplet again
                    used.add(required);
                }
 
                set.add(arr[j]);
            }
        }
 
        return ans;
