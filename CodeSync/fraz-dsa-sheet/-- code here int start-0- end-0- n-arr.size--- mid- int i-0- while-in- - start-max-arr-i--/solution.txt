    private boolean canSplit(int[] arr, int k, int maxSum) {
        int count = 1;
        int sum = 0;

        for (int x : arr) {
            if (sum + x > maxSum) {
                count++;
                sum = x;
            } else {
                sum += x;
            }
        }

        return count <= k;
    }
