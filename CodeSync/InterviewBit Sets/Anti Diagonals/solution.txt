public class Solution {
    public int[][] diagonal(int[][] A) {

        int n = A.length;
        int[][] ans = new int[2 * n - 1][n];

        for (int d = 0; d < 2 * n - 1; d++) {

            int row, col;

            if (d < n) {
                row = 0;
                col = d;
            } else {
                row = d - n + 1;
                col = n - 1;
            }

            int k = 0;

            while (row < n && col >= 0) {
                ans[d][k++] = A[row][col];
                row++;
                col--;
            }
        }

        return ans;
    }
}
