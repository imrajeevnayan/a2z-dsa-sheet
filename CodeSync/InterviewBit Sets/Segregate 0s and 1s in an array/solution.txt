int left = 0;
        int right = A.length - 1;
        
        while (left < right) {
            // Agar left pe 0 hai, toh woh already sahi jagah hai, bas aage badho
            if (A[left] == 0) {
                left++;
            } 
            // Agar right pe 1 hai, toh woh already sahi jagah hai, bas peeche aao
            else if (A[right] == 1) {
                right--;
            } 
            // Agar left pe 1 hai aur right pe 0 hai, toh swap karo
            else {
                // Swap A[left] and A[right]
                int temp = A[left];
                A[left] = A[right];
                A[right] = temp;
                left++;
                right--;
            }
        }
        
        return A;
    }
