public class Solution {
    public int maxProduct(final int[] A) {

        int max = A[0];
        int min = A[0];
        int ans = A[0];

        for (int i = 1; i < A.length; i++) {
            int num = A[i];

            int newMax = Math.max(num, Math.max(num * max, num * min));
            int newMin = Math.min(num, Math.min(num * max, num * min));

            max = newMax;
            min = newMin;

            ans = Math.max(ans, max);
        }

        return ans;
    }
}
