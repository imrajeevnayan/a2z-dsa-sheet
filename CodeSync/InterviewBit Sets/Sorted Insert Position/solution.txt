public class Solution {
    public int searchInsert(ArrayList<Integer> a, int b) {
        int low = 0;
        int high = a.size() - 1;
        int ans = a.size(); // Default: Agar sab chote hain, toh end mein insert hoga
        
        while (low <= high) {
            int mid = low + (high - low) / 2;
            
            if (a.get(mid) == b) return mid; // Exact match mil gaya
            else if (a.get(mid) > b) {
                ans = mid; // Ye potential position hai, store karo
                high = mid - 1; // Ab aur chota dhundo left side mein
            } else 
                low = mid + 1; // Right side mein dhundo
            
        }
        
        return ans;
    }
    
}
