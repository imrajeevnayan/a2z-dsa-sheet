for (int i = 0; i <= A.length; i++) {

            if (i < A.length && A[i] >= 0) {
                currSum += A[i];
            } else {

                int currLen = i - start;
                int bestLen = bestEnd - bestStart + 1;

                if (currSum > maxSum ||
                    (currSum == maxSum && currLen > bestLen)) {

                    maxSum = currSum;
                    bestStart = start;
                    bestEnd = i - 1;
                }

                currSum = 0;
                start = i + 1;
            }
        }

        int[] ans = new int[bestEnd - bestStart + 1];

        for (int i = bestStart; i <= bestEnd; i++) {
            ans[i - bestStart] = A[i];
        }

        return ans;
    }
