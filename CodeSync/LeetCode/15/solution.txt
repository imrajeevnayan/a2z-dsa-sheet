int slidingwindow(vector<int>& nums, int desired){
        unordered_map<int,int> mp;
        int l =0;
        int totalcount = 0;
        for(int r = 0; r<nums.size(); r++){
            //EXPAND window(r)
            mp[nums[r]]++; //some code
            
            //INVALID state
            while(mp.size() > desired){
                mp[nums[l]] --;
                if(mp[nums[l]] == 0)
                    mp.erase(nums[l]);
                l++; // SHRINK window(l) 
            }
            totalcount += (r - l +1); //CALCULATE ans
        }
        return totalcount;
    }
