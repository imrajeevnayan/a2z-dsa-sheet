4        boolean[] ans = new boolean[strs.length];
5        int[] pS = new int[n + 1]; 
6        for (int i = 0; i < n; i++) {

8        }
9        for (int i = 0; i < strs.length; i++) {

11        }
12        return ans;
13    }
14    static boolean check(String s, String t, int[] pS, int n) {
15        int ones = pS[n], fixed = 0;
16        char[] arr = t.toCharArray();
17        List<Integer> qPos = new ArrayList<>();
18        for (int j = 0; j < n; j++) {
19            if (arr[j] == '1') fixed++;
20            else if (arr[j] == '?') {
21                arr[j] = '0';
22                qPos.add(j);
23            }
24        }
25        int need = ones - fixed;
26        if (need < 0 || need > qPos.size()) return false;
27        for (int j = qPos.size() - 1; j >= qPos.size() - need; j--) {
28            arr[qPos.get(j)] = '1';
29        }  
30        int[] pT = new int[n + 1];
31        for (int j = 0; j < n; j++) {
32            pT[j + 1] = pT[j] + (arr[j] - '0');
33        }
34        for (int i = 0; i <= n; i++) {
35            if (pS[i] < pT[i]) return false;
36        }
37        return true;
38    }
