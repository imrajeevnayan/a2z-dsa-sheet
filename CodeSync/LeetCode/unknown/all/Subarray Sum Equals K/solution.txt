class Solution {
      public int subarraySum(int[] arr, int k) {
        Map<Integer, Integer> map = new HashMap<>();
          map.put(0, 1);
         int sum = 0, count = 0;
          for (int n : arr) {
                 sum += n;
            if(map.containsKey(sum-k)) count+=map.get(sum-k);
            map.put(sum,map.getOrDefault(sum,0)+1); 
         }
         return count;
    }
}
