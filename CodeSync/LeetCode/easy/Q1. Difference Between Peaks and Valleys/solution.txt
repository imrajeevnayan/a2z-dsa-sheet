class Solution {
    public int peakValleyDifference(int[] nums) {


        int peakSum = 0;
        int valleySum = 0;


        // First aur last index peak/valley nahi ho sakte
        for (int i = 1; i < nums.length - 1; i++) {


            // Peak check karo
            if (nums[i] > nums[i - 1] &&
                nums[i] > nums[i + 1]) {


                peakSum += nums[i];
            }


            // Valley check karo
            else if (nums[i] < nums[i - 1] &&
                     nums[i] < nums[i + 1]) {


                valleySum += nums[i];
            }
        }


        // Peak values ka sum - Valley values ka sum
        return peakSum - valleySum;
    }
}
