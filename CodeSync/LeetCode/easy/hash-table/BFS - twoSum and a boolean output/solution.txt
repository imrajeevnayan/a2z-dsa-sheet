class Solution {
    public boolean findTarget(TreeNode root, int k) {
        if(root==null)return false;
    }
        Set<Integer>set=new HashSet<>();
        if(set.contains(k-root.data))return true;
        set.add(root.data);
        return findTarget(TreeNode root, int k)
}
