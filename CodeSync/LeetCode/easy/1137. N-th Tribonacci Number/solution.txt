class Solution {
    private int solve(int n, int[] dp) {
        // Base cases
        if (n == 0) return 0;  
        if (n == 1 || n == 2) return 1;
        // Already calculated
        if (dp[n] != -1)  return dp[n];
        

        // Store the result
        dp[n] = solve(n - 1, dp)
              + solve(n - 2, dp)
              + solve(n - 3, dp);

        return dp[n];
    }
    public int tribonacci(int n) {
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        return solve(n, dp);
    }
}
