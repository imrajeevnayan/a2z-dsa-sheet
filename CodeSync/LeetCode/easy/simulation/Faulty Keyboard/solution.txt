            if (ch == 'i') {
                reversed = !reversed; // toggle direction
            } else {
                if (!reversed) {
                    sb.append(ch);
                } else {
                    sb.insert(0, ch); // simulate reversed insertion
                }
            }
        }

        if (reversed) {
            sb.reverse();
        }

        return sb.toString();
    }
