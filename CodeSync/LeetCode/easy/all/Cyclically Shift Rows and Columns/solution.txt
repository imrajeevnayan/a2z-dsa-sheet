class Solution {
    public int[][] cyclicShift(int n, int[][] grid, int[] rowShift, int[] colShift) {
        // Step 1: Har row ko LEFT shift karo
        int[][] temp = new int[n][n];


        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int newCol = (j - rowShift[i] + n) % n;
                temp[i][newCol] = grid[i][j];
            }
        }
        // Step 2: Har column ko UPWARD shift karo
        int[][] ans = new int[n][n];


        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int newRow = (i - colShift[j] + n) % n;
                ans[newRow][j] = temp[i][j];
            }
        }
        return ans;
    }
}
