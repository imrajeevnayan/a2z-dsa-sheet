for (int i = 0; i < num.length(); i++) {
            int digit = num.charAt(i) - '0';
            if (i % 2 == 0)  evenSum += digit;
            else   oddSum += digit;
        }
        int evenSum = 0,oddSum = 0;
class Solution {
    public boolean isBalanced(String num) {
        return evenSum == oddSum;
    }
}
