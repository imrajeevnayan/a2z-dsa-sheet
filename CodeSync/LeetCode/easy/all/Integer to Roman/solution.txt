2    private static final int[]    V = {1000,900,500,400,100,90,50,40,10,9,5,4,1};
3    private static final String[] S = {"M","CM","D","CD","C","XC","L","XL","X","IX","V","IV","I"};
4    public String intToRoman(int num) {
5        StringBuilder sb = new StringBuilder();
6        for (int i = 0; i < V.length; i++) {
7            while (num >= V[i]) {
8                sb.append(S[i]);
9                num -= V[i];
10            }
11        }
12        return sb.toString();
13    }
