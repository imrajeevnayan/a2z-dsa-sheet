class Solution {
    public int countIntersectingIntervals(int[][] intervals) {
        int n = intervals.length;
        Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0]));
        
        List<Integer> ends = new ArrayList<>();
        long nonIntersecting = 0;
        
        for (int[] interval : intervals) {
            int start = interval[0];
            
            // Binary search: count previous ends < current start
            int lo = 0, hi = ends.size();
            while (lo < hi) {
                int mid = (lo + hi) >>> 1;
                if (ends.get(mid) < start) lo = mid + 1;
                else hi = mid;
            }
            nonIntersecting += lo;
            
            // Insert current end in sorted position
            int pos = Collections.binarySearch(ends, interval[1]);
            if (pos < 0) pos = -(pos + 1);
            ends.add(pos, interval[1]);
        }
        
        long totalPairs = (long) n * (n - 1) / 2;
        return (int) (totalPairs - nonIntersecting);
    }
}
