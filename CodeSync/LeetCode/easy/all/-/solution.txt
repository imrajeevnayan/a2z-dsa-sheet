class Solution {
    public boolean isHappy(int n) {
        Set<Integer>s=new HashSet<>();
        while(n!=1 && ! s.contains(n)){
                s.add(n);
                n=getNext(n);
        }
        return n==1;
    }
    static int getNext(int n){
        int totalSum=0;
        while(n > 0){
            int d=n %10;
            totalSum+=d*d;
            n/=10;
        }
        return totalSum;
    }
}
