class Solution {
    public long maxPairStrength(int[] nums) {
        int n = nums.length;
        long maxStrength = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int g = gcd(nums[i], nums[j]);
                long strength = (long)(nums[i] / g) * (nums[j] / g);
                if (strength > maxStrength) {
                    maxStrength = strength;
                }
            }
        }
        return maxStrength;
    }
    
    private int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
