class Solution {
    public int change(int amount, int[] coins) {
        return solve(0, amount, coins);
    }
    private int solve(int i, int amount, int[] coins) {
        if (amount == 0) return 1;

        // Base Case 2: Out of bounds (invalid amount or ran out of coins)
        if (amount < 0 || i >= coins.length) return 0;

        // Choice 1: Take the coin at current index (stay at index 'i' for unlimited supply)
        int take = solve(i, amount - coins[i], coins);

        // Choice 2: Skip the coin at current index (move to index 'i + 1')
        int skip = solve(i + 1, amount, coins);

        // Total ways is the sum of both choices
        return take + skip;
    }
}
