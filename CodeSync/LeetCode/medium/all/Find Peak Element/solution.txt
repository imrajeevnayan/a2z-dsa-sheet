class Solution {
    public int findPeakElement(int[] nums) {
        int left = 0,right = nums.length - 1,ans=-1;
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] < nums[mid + 1]) left=mid
            else  left = mid + 1;
        }
        return left;
    }
}
