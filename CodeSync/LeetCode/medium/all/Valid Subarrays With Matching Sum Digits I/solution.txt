class Solution {
    public int countValidSubarrays(int[] nums, int x) {
        int n = nums.length;
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            int v = nums[i];
            int ds = 0;
            while (v > 0) {
                ds += v % 10;
                v /= 10;
            }
            arr[i] = nums[i] - ds * x;
        }
        int ans = 0;
        int pref = 0;
        java.util.HashMap<Integer, Integer> map = new java.util.HashMap<>();
        map.put(0, 1);
        for (int val : arr) {
            pref += val;
            ans += map.getOrDefault(pref, 0);
            map.put(pref, map.getOrDefault(pref, 0) + 1);
        }
        return ans;
    }
}
