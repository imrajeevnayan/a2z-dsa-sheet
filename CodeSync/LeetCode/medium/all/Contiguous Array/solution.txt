for(int i=0;i<nums.length;i++){
            sum+=(nums[i]==0) ? -1 :1;
           if (map.containsKey(sum)) {
                int prev = map.get(sum);
                maxLen = Math.max(maxLen, i - prev);
            }
             else  map.put(sum, i);
        }
        int maxLen = 0,sum = 0;
        return maxLen;
    }
