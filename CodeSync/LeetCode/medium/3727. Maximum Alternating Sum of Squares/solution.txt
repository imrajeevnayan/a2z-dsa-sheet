Arrays.sort(nums);
        int n = nums.length;
        long sum = 0;
        boolean isPositive = true; // Pehla term positive hoga
        
        int left = 0;
        int right = n - 1;
        
        while (left <= right) {
            if (isPositive) {
                // Largest available number lo
                sum += (long) nums[right] * nums[right];
                right--;
            } else {
                // Smallest available number lo
                sum -= (long) nums[left] * nums[left];
                left++;
            }
            isPositive = !isPositive; // Sign flip karo
        }
        
        return sum;
    }
