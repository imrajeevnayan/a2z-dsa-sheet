// Initialize answer with all bits set (equivalent to -1 in two's complement for int)
        // We will AND this with relevant numbers.
        int ans = -1; 
        boolean foundMisplaced = false;
        
        for (int i = 0; i < n; i++) {
            if (nums[i] != i) {
                foundMisplaced = true;
                // Include the value currently at this index
                ans &= nums[i];
                // Include the value that SHOULD be at this index (which is i)
                ans &= i;
            }
        }
        
        // If the array is already sorted, problem states return 0
        if (!foundMisplaced) {
            return 0;
        }
        
        return ans;
    }
