class Solution {
    public int findTheWinner(int n, int k) {
    while(q.size() > 1){
    for(int i = 1; i < k; i++){
        q.offer(q.poll());
    }
    q.poll();
}
    }
    Queue<Integer>q=new Queue<>();
    for(int i: n)q.add(i);
}
