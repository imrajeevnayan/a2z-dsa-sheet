class Solution {
    public int maxDistance(String moves) {
         int x = 0, y = 0;
        for (char c : moves.toCharArray()) {
            if (c == 'E') x++;
            else if (c == 'W') x--;
            else if (c == 'N') y++;
            else if (c == 'S') y--;
        }
        return Math.abs(x) + Math.abs(y);
    }
}
