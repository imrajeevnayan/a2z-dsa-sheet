    private boolean canBeSortedInOneMove(char[] current, char[] target) {
        if (Arrays.equals(current, target)) return true;
        
        int left = -1, right = -1;
        for (int i = 0; i < current.length; i++) {
            if (current[i] != target[i]) {
                if (left == -1) left = i;
                right = i;
            }
        }
        if (left == -1) return true; // Already matched
        
        char[] temp = Arrays.copyOf(current, current.length);
        Arrays.sort(temp, left, right + 1);
        return Arrays.equals(temp, target);
    }
