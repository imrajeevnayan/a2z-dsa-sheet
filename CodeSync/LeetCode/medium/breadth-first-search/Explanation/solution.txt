class Solution {
    public int findCircleNum(int[][] isConnected) {
        int n = isConnected.length;
        boolean[] visited = new boolean[n];
        int provinces = 0;
        for (int i = 0; i < n; i++) {

            if (!visited[i]) {
                provinces++;
                bfs(i, isConnected, visited);
            }
        }
        return provinces;
    }
