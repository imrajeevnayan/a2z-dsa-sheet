public int[] findThePrefixCommonArray(int[] A, int[] B) {
        int n = A.length;
        int[] ans = new int[n];
        int[] seen = new int[n + 1];

        int common = 0;

        for (int i = 0; i < n; i++) {
            if (++seen[A[i]] == 2) common++;

            if (++seen[B[i]] == 2)
                common++;

            ans[i] = common;
        }

        return ans;
    }
