class Solution {
    int solve(int i, int[] nums, int end,int[]dp) {
        if (i > end) return 0;
        
        int take = nums[i] + solve(i + 2, nums, end);
        int skip = solve(i + 1, nums, end);
        return Math.max(take, skip);
    }
