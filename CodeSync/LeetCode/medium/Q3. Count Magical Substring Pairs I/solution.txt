class Solution {
    public int countMagicalPairs(String s) {
        int n = s.length();
        int[][] distinct = new int[n][n];
        // Precompute distinct characters for every substring
        for (int i = 0; i < n; i++) {


            int[] freq = new int[26];
            int cnt = 0;


            for (int j = i; j < n; j++) {


                int ch = s.charAt(j) - 'a';


                if (freq[ch] == 0)
                    cnt++;


                freq[ch]++;


                distinct[i][j] = cnt;
            }
        }


        long ans = 0;


        // Fix partition point j
        for (int j = 0; j < n - 1; j++) {


            int[] count = new int[27];


            // All left substrings ending at j
            for (int i = 0; i <= j; i++) {
                count[distinct[i][j]]++;
            }


            // All right substrings starting at j+1
            for (int k = j + 1; k < n; k++) {
                ans += count[distinct[j + 1][k]];
            }
        }


        return (int) ans;
    }
}
