        int[] dp = new int[subsetSum + 1];
        dp[0] = 1; // There is one way to get sum 0: choose nothing
        
        for (int num : nums) {
            for (int j = subsetSum; j >= num; j--) {
                dp[j] += dp[j - num];
            }
        }
