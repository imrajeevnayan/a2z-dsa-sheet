    private void dfs(int[][] grid, int r, int c, boolean[][] vis) {
        if (r < 0 || c < 0 || r >= m || c >= n ||
            grid[r][c] == 0 || vis[r][c]) {
            return;
        }

        vis[r][c] = true;
        sum += grid[r][c];

        dfs(grid, r + 1, c, vis);
        dfs(grid, r - 1, c, vis);
        dfs(grid, r, c + 1, vis);
        dfs(grid, r, c - 1, vis);
    }
