import java.util.*;

class Solution {
    public List<Integer> pancakeSort(int[] arr) {
        List<Integer> result = new ArrayList<>();
        int n = arr.length;
        
        // Hum end se start karenge, har step pe largest element ko fix karenge
        for (int currSize = n; currSize > 1; currSize--) {
            // Step 1: Find index of the maximum element in arr[0...currSize-1]
            int maxIndex = findMaxIndex(arr, currSize);
            
            // If max element is already at the end, no need to flip
            if (maxIndex == currSize - 1) {
                continue;
            }
            
            // Step 2: Flip to bring max element to the front (index 0)
            if (maxIndex != 0) {
                flip(arr, maxIndex + 1);
                result.add(maxIndex + 1);
            }
            
            // Step 3: Flip to move max element to its correct position (currSize - 1)
            flip(arr, currSize);
