while (!dq.isEmpty() && nums[dq.peekLast()] < nums[right]) {
                int r = dq.pollLast();
                int l = dq.isEmpty() ? left - 1 : dq.peekLast();

                ops -= (long) (r - l) * (nums[right] - nums[r]);
            }

            dq.offerLast(right);

            while (ops < 0) {
                ops += nums[dq.peekFirst()] - nums[left];

                if (dq.peekFirst() == left) {
                    dq.pollFirst();
                }

                left++;
            }

            ans += right - left + 1;
        }

        return ans;
