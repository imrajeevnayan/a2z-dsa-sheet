            while (cost > k) {
                cost -= nums[stack.peek()] - nums[r];
                if (stack.peek() == r) {
                    stack.pop();
                    if (!stack.isEmpty()) {
                        int first = stack.pop();

                        int next = stack.isEmpty() ? r + 1 : stack.peek();

                        cost -= (long) (next - first) *
                                (nums[first] - nums[r]);

                        stack.push(first);
                    }
                }
                r--;
            }

            ans += r - l + 1;
        }
