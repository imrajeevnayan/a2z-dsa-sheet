    private long getHash(int l, int r) {
        int len = r - l + 1;

        long h1 = (pref1[r + 1]
                - (pref1[l] * pow1[len]) % MOD1
                + MOD1) % MOD1;

        long h2 = (pref2[r + 1]
                - (pref2[l] * pow2[len]) % MOD2
                + MOD2) % MOD2;

        return (h1 << 32) ^ h2;
