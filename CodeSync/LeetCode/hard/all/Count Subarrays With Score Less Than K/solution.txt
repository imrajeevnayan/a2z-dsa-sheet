class Solution {
    public long countSubarrays(int[] nums, long k) {
        long count = 0, sum = 0;
        int left = 0;
        for (int right = 0; right < nums.length; right++) {
            sum += nums[right];
            int len=right-left+1;
            while (sum * (right-left+1)>= k) {
                sum -= nums[left];
                left++;
            }
            count += right-left+1;
        }
        return count;
    }
}
