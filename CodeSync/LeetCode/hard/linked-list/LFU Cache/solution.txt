    private void update(Node node) {
        int freq = node.freq;
        LinkedHashSet<Node> set = freqMap.get(freq);

        set.remove(node);

        if (freq == minFreq && set.isEmpty()) {
            minFreq++;
        }

        node.freq++;

        freqMap.computeIfAbsent(node.freq, k -> new LinkedHashSet<>()).add(node);
    }
