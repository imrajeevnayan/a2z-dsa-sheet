class Solution{
    static int solve(int sum, int index, int[] coins, int[][] dp) {
        if (sum == 0) return 1;
        if (index >= coins.length) return 0;

        if (dp[index][sum] !=-1) return dp[index][sum];
        
        int take = 0;
        if (coins[index] <= sum) {
            take = solve(sum - coins[index], index, coins, dp);
        }

        // Skip current coin
        int skip = solve(sum, index + 1, coins, dp);
        return dp[index][sum] = take + skip;
    }
    public int count(int coins[], int sum) {
        int[][] dp = new int[coins.length][sum + 1];
        for (int[] row : dp) {
            Arrays.fill(row, -1);
        }
        return solve(sum, 0, coins, dp);
    }
}
