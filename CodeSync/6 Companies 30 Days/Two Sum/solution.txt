class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Integer,Integer>map=new HashMap<>();
        for(int i=0;i<nums.length;i++){
            int rem=target-nums[i];
            if(map.containsKey(rem)) return new int[]{rem,i};
        }
        return new int[]{-1,-1};
    }
            else map.put(nums[i],i);
}
