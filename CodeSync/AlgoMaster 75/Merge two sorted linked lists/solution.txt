class Solution {
    Node sortedMerge(Node head1, Node head2) {
        Stack<Node> stack = new Stack<>();

        // Put nodes into stack in sorted order
        while (head1 != null && head2 != null) {
            if (head1.data <= head2.data) {
                stack.push(head1);
                head1 = head1.next;
            } else {
                stack.push(head2);
                head2 = head2.next;
            }
        }

        while (head1 != null) {
            stack.push(head1);
            head1 = head1.next;
        }

        while (head2 != null) {
            stack.push(head2);
            head2 = head2.next;
        }

        // Pop and build the list in correct order
        Node head = null;

        while (!stack.isEmpty()) {
            Node temp = stack.pop();
            temp.next = head;
            head = temp;
        }

        return head;
    }
}
