public class Solution {  
    
    public static int duplicateNumber(int arr[]) {
        
        int ans = 0;
 
        // Array ke saare elements ka XOR
        for (int i = 0; i < arr.length; i++) {
            ans = ans ^ arr[i];
        }
 
        // 0 se n-2 tak ke numbers ka XOR
        for (int i = 0; i < arr.length - 1; i++) {
            ans = ans ^ i;
        }
 
        return ans;
    }
}
