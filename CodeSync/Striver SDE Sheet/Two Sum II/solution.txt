import java.util.* ;
import java.io.*; 
public class Solution {
    public static int[] twoSum(int numbers[], int target) {
        int left = 0, right = numbers.length - 1;
 
        while (left < right) {
            int sum = numbers[left] + numbers[right];
 
            if (sum == target) {
                return new int[]{left, right};
            } else if (sum < target) {
                left++;
            } else {
                right--;
            }
        }
        return new int[]{-1, -1}; // if no pair found
    }
}
