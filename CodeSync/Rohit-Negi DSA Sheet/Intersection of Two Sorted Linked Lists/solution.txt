class Solution {
    public Node findIntersection(Node head1, Node head2) {

        Set<Integer> set = new HashSet<>();

        // head1 ki values store karo
        while (head1 != null) {
            set.add(head1.data);
            head1 = head1.next;
        }

        Node dummy = new Node(-1);
        Node tail = dummy;

        // head2 ki common values check karo
        while (head2 != null) {

            if (set.contains(head2.data)) {
                tail.next = new Node(head2.data);
                tail = tail.next;
            }

            head2 = head2.next;
        }

        return dummy.next;
    }
}
