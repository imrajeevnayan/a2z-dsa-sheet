class Solution {
    public int getMinimumDays(String s, int[] p) {
        int n = s.length();

        int[] day = new int[n];
        for (int i = 0; i < n; i++) {
            day[p[i]] = i + 1; // removal day of each index
        }

        int low = 0, high = n;
        int ans = n;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (isValid(s, day, mid)) {
                ans = mid;
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }

        return ans;
    }

    private boolean isValid(String s, int[] day, int days) {
        int n = s.length();

        for (int i = 0; i < n - 1; i++) {
            if (s.charAt(i) == s.charAt(i + 1)) {

                // both characters still exist after 'days'
                if (day[i] > days && day[i + 1] > days) {
                    return false;
                }
            }
        }

        return true;
    }
}
