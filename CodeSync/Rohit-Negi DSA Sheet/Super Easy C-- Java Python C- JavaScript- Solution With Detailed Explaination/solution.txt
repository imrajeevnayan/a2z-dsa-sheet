class Solution {
    static int solve(int n) {
        if (n == 0) return 0;
        int count = containsFour(n) ? 1 : 0;
        return count + solve(n - 1);
    }

    static boolean containsFour(int num) {

        while (num > 0) {

            if (num % 10 == 4)
                return true;

            num /= 10;
        }

        return false;
    }

    public static int countNumberswith4(int n) {

        return solve(n);
    }
}
