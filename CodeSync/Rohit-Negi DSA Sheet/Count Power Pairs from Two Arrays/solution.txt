class Solution {

    private int upperBound(int[] b, int x) {
        int low = 0, high = b.length;

        while (low < high) {
            int mid = low + (high - low) / 2;

            if (b[mid] <= x)
                low = mid + 1;
            else
                high = mid;
        }

        return low;
    }

    private long countForX(int x, int[] b, int[] cnt) {

        if (x == 0)
            return 0;

        if (x == 1)
            return cnt[0];

        int idx = upperBound(b, x);

        long ans = b.length - idx;

        ans += cnt[0] + cnt[1];

        if (x == 2) {
            ans -= (cnt[3] + cnt[4]);
        }

        if (x == 3) {
            ans += cnt[2];
        }

        return ans;
    }
