class Solution {
    public static String minWindow(String s, String p) {
        if (s == null || p == null || s.length() < p.length()) {
            return "";
        }

        // Step 1: Store frequency of characters in p
        HashMap<Character, Integer> freqP = new HashMap<>();
        for (char c : p.toCharArray()) {
            freqP.put(c, freqP.getOrDefault(c, 0) + 1);
        }

        int required = freqP.size(); // Kitne unique characters match karne hain
        int formed = 0; // Kitne characters abhi tak satisfy ho chuke hain

        // Window ke andar ki frequencies
        HashMap<Character, Integer> windowFreq = new HashMap<>();

        int left = 0, right = 0;
        int minLen = Integer.MAX_VALUE;
        int startIndex = 0;

        while (right < s.length()) {
            char c = s.charAt(right);
            windowFreq.put(c, windowFreq.getOrDefault(c, 0) + 1);

            // Agar current character 'p' mein hai aur uski frequency match ho gayi
            if (freqP.containsKey(c) && windowFreq.get(c).intValue() == freqP.get(c).intValue()) {
                formed++;
            }

            // Step 2: Try to shrink the window from the left
            while (left <= right && formed == required) {
                char startChar = s.charAt(left);

                // Update minimum window
                if (right - left + 1 < minLen) {
                    minLen = right - left + 1;
                    startIndex = left;
                }

                // Remove character from window
