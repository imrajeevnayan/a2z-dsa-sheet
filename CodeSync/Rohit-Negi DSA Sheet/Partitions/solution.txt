        }

        long target = total / 3;

        long prefixSum = 0;
        long countTarget = 0;
        long ways = 0;

        // last element cannot be a cut point
        for (int i = 0; i < A - 1; i++) {

            prefixSum += B[i];

            if (prefixSum == 2 * target) {
                ways += countTarget;
            }

            if (prefixSum == target) {
                countTarget++;
            }
        }

        return (int) ways;
    }
