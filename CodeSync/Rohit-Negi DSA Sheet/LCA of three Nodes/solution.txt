import java.util.*;
class Solution {
    public static int KthLargest(TreeNode<Integer> root, int k) {
        PriorityQueue<Integer> pq = new PriorityQueue<>(Collections.reverseOrder());
        store(root, pq);
        if (pq.size() < k) return -1;
        
        // K-1 largest elements remove karo
        for (int i = 1; i < k; i++) {
            pq.poll();
        }
        return pq.peek();
    }
    private static void storeTreeNode<Integer> root,
            PriorityQueue<Integer> pq) {
 
        if (root == null) {
            return;
        }
 
        pq.offer(root.data);
 
        store(root.left, pq);
        store(root.right, pq);
    }
}
