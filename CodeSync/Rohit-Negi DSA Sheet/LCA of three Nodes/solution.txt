import java.util.*;
 
class Solution {
 
    public static int KthLargest(TreeNode<Integer> root, int k) {
 
        PriorityQueue<Integer> pq = new PriorityQueue<>();
 
        inorder(root, pq, k);
 
        // K-th largest doesn't exist
        if (pq.size() < k) {
            return -1;
        }
 
        return pq.peek();
    }
 
    private static void inorder(
            TreeNode<Integer> root,
            PriorityQueue<Integer> pq,
            int k) {
 
        if (root == null) {
            return;
        }
 
        inorder(root.left, pq, k);
 
        pq.offer(root.data);
 
        // Keep only K largest elements
        if (pq.size() > k) {
            pq.poll();
        }
 
        inorder(root.right, pq, k);
    }
}
