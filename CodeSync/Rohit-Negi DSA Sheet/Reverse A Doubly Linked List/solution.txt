public class Solution {
    public static Node reverseDLL(Node head) {
        if (head == null) return null;
        
        Node curr = head;
        Node newHead = null;
 
        while (curr != null) {
            Node temp = curr.prev;
            curr.prev = curr.next;
            curr.next = temp;
 
            newHead = curr;
 
            // Move to old next
            curr = curr.prev;
        }
 
        return newHead;
    }
}
