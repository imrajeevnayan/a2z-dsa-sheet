class Solution {
    int unvisitedLeaves(int arr[], int k) {
        boolean[] vis = new boolean[k + 1];
        
        Set<Integer> set = new  HashSet<>();

        // Duplicates remove karo
        for (int s : arr) set.add(s);

        // Multiples mark karo
        for (int s : set) {
            if (s > k) continue;
            for (int i = s; i <= k; i += s) {
                vis[i] = true;
            }
        }

        // Unvisited count karo
        int cnt = 0;
        for (int i = 1; i <= k; i++) {
            if (!vis[i]) cnt++;
        }

        return cnt;
    }
}
