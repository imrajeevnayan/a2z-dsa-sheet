            StringBuilder sb = new StringBuilder();

            // Last line or single word
            if (j == words.length || gaps == 0) {
                for (int k = i; k < j; k++) {
                    if (k > i) sb.append(" ");
                    sb.append(words[k]);
                }
            } else {
                int space = (l - len) / gaps;
                int extra = (l - len) % gaps;

                for (int k = i; k < j; k++) {
                    sb.append(words[k]);

                    if (k < j - 1) {
                        sb.append(" ".repeat(space + (extra-- > 0 ? 1 : 0)));
                    }
                }
            }

            // Add trailing spaces
            while (sb.length() < l)
                sb.append(" ");

            ans.add(sb.toString());
            i = j;
        }
