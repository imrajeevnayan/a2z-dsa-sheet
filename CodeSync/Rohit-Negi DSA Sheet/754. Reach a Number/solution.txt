class Solution {
    public int reachNumber(int target) {
        // 1. Handle symmetry: work with positive target
        target = Math.abs(target);
        
        int sum = 0;
        int steps = 0;
        
        // 2. Keep adding steps until we meet two conditions:
        //    a) sum >= target (we've reached or passed it)
        //    b) (sum - target) is even (we can flip some moves to adjust)
        while (true) {
            if (sum >= target && (sum - target) % 2 == 0) {
                return steps;
            }
            
            steps++;
            sum += steps;
        }
    }
}
