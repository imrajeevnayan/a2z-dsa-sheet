class Solution {
    public int firstSearch(int[] arr, int k) {
         int left = 0,right = arr.length - 1,ans = -1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (arr[mid] == k) {
                ans = mid;       // store answer
                right = mid - 1; // search left side
            }
            else if (arr[mid] < k) left = mid + 1;
            else right = mid - 1;
        }
        return ans;
    }
}
