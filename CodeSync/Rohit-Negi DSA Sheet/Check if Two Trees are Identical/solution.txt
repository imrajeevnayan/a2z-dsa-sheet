    // Function to check if two trees are identical
    boolean isIdentical(Node r1, Node r2)
    {
        // If both trees are empty, they are identical
        if (r1 == null && r2 == null)
            return true;
        // If only one tree is empty, they are not identical
        if (r1 == null || r2 == null)
            return false;
        // Check data and compare left and right subtrees
        return (r1.data == r2.data)
            && isIdentical(r1.left, r2.left)
            && isIdentical(r1.right, r2.right);

    // Stores the information of the next node in Morris
    // preorder traversal
    static class Result {
        boolean found;
        int data;
        boolean hasLeft;
        boolean hasRight;
​
        Result(boolean found, int data, boolean hasLeft,
               boolean hasRight)
        {
            this.found = found;
            this.data = data;
            this.hasLeft = hasLeft;
            this.hasRight = hasRight;
        }
    }
​
    // Returns the next node in Morris preorder traversal
    // and stores its data and child structure
    static Result getNext(Node[] curr)
    {
​
        while (curr[0] != null) {
​
            // If there is no left subtree
            if (curr[0].left == null) {
                int data = curr[0].data;
                boolean hasLeft = false;
                boolean hasRight = curr[0].right != null;
​
                curr[0] = curr[0].right;
​
                return new Result(true, data, hasLeft,
                                  hasRight);
            }
​
            // Find the inorder predecessor
            Node pred = curr[0].left;
​
            while (pred.right != null
                   && pred.right != curr[0])
                pred = pred.right;
​
            // Create a temporary thread
            if (pred.right == null) {
                int data = curr[0].data;
                boolean hasLeft = true;
                boolean hasRight = curr[0].right != null;
​
                pred.right = curr[0];
                curr[0] = curr[0].left;
​
                return new Result(true, data, hasLeft,
                                  hasRight);
            }
​
            // Remove the temporary thread
            pred.right = null;
            curr[0] = curr[0].right;
        }
​
        return new Result(false, 0, false, false);
    }
​
    static boolean isIdentical(Node r1, Node r2)
    {
​
        Node[] curr1 = { r1 };
        Node[] curr2 = { r2 };
​
        while (true) {
​
            Result res1 = getNext(curr1);
            Result res2 = getNext(curr2);
​
            // If both traversals are complete
            if (!res1.found && !res2.found)
                return true;
​
            // If only one traversal is complete
            if (res1.found != res2.found)
                return false;
​
            // Compare node data and tree structure
            if (res1.data != res2.data
                || res1.hasLeft != res2.hasLeft
                || res1.hasRight != res2.hasRight)
                return false;
        }
    }
