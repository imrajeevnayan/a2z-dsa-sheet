class Solution {
    public int minDiff(Node root, int k) {

        // minimum difference store karega
        int ans = Integer.MAX_VALUE;

        Node curr = root;

        while (curr != null) {

            // current node ke saath difference calculate karo
            ans = Math.min(ans, Math.abs(curr.data - k));

            // exact match mil gaya
            if (curr.data == k) {
                return 0;
            }

            // BST property use karo
            if (k < curr.data) {
                curr = curr.left;   // left subtree me jao
            } else {
                curr = curr.right;  // right subtree me jao
            }
        }

        return ans;
    }
}
