public class Solution {
    public static void deleteAlternateNodes(Node<Integer> head) {
        // Edge case: Agar list empty hai ya sirf 1 node hai, toh kuch delete nahi hoga
        if (head == null || head.next == null)  return;
        
        Node<Integer> current = head;
 
        // Jab tak current aur uska next dono exist karte hain
        while (current != null && current.next != null) {
            // Alternate node (current.next) ko bypass karo
            Node<Integer> nodeToDelete = current.next;
            current.next = current.next.next;
 
            // Optional: Memory leak avoid karne ke liye reference null karo
            // (Java GC handle kar lega, lekin good practice hai)
            nodeToDelete.next = null; 
 
            // Ab 'current' ko agle surviving node par move karo
            // Kyunki current.next ab already updated ho chuka hai
            current = current.next;
        }
    }
}
