import java.util.Scanner;
 
public class Main {
 
    static int n, m;
    static int[][] grid;
 
    static int[] dr = {-1, 0, 1, 0};
    static int[] dc = {0, 1, 0, -1};
 
    public static void dfs(int r, int c) {
 
        grid[r][c] = 0; // mark visited
 
        for (int k = 0; k < 4; k++) {
            int nr = r + dr[k];
            int nc = c + dc[k];
 
            if (nr >= 0 && nr < n &&
                nc >= 0 && nc < m &&
                grid[nr][nc] == 1) {
 
                dfs(nr, nc);
            }
        }
    }
 
    public static void main(String[] args) {
 
        Scanner sc = new Scanner(System.in);
 
        n = sc.nextInt();
        m = sc.nextInt();
 
        grid = new int[n][m];
 
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = sc.nextInt();
            }
        }
