class Solution {
    int maxSubarraySum(int[] arr) {
        int maxSum=Integer.MIN_VALUE,curr=0;
        for(int n :arr){
            curr+=n;
            maxSum=Math.max(maxSum,curr);
            if(curr<0)curr=0;
        }
        return maxSum;
    }
}
