import java.util.*;
import java.io.*;
 
public class Solution {
    public static boolean checkXORPairs(int[] arr, int n) {
 
        HashMap<Integer, Integer> map = new HashMap<>();
 
        for (int i = 0; i < n; i++) {
            map.put(arr[i], map.getOrDefault(arr[i], 0) + 1);
        }
 
        for (int freq : map.values()) {
            if (freq % 2 != 0) {
                return false;
            
        }
 
        return true;
    }
}
