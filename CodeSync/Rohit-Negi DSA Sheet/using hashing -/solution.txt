class Solution {
    public Node findIntersection(Node head1, Node head2) {

        // List 1 ke saare nodes Set mein store karo
        HashSet<Node> set = new HashSet<>();

        Node curr = head1;

        while (curr != null) {
            set.add(curr);
            curr = curr.next;
        }

        // List 2 traverse karo
        curr = head2;

        while (curr != null) {

            // Agar same node List 1 mein bhi hai,
            // to wahi intersection node hai
            if (set.contains(curr)) {
                return curr;
            }

            curr = curr.next;
        }

        // Intersection nahi mila
        return null;
    }
}
