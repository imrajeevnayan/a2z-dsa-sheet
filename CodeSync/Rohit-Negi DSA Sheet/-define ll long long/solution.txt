class Solution {
    public int countSubarray(int[] arr, int k) {

        long mod = 1000000007L;
        int n = arr.length;

        // Total possible subarrays
        long totalSubarrays = (long) n * (n + 1) / 2;

        long validSubarrays = 0;

        int len = 0;

        for (int i = 0; i < n; i++) {

            // Current element k se chhota ya equal hai
            if (arr[i] <= k) {
                len++;
            }
            else {

                // Current continuous segment ka contribution
                validSubarrays += (long) len * (len + 1) / 2;

                // Naya segment start hoga
                len = 0;
            }
        }

        // Last segment add karna na bhoolo
        validSubarrays += (long) len * (len + 1) / 2;

        // Max > k wale subarrays
        long ans = totalSubarrays - validSubarrays;

        return (int) (ans % mod);
    }
}
