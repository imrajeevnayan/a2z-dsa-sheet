public class Solution {
 
    public static boolean detectCycle(Node<Integer> head) {
 
        Node<Integer> slow = head;
        Node<Integer> fast = head;
 
        while (fast != null && fast.next != null) {
 
            slow = slow.next;           // 1 step
            fast = fast.next.next;      // 2 steps
 
            if (slow == fast) {
                return true;
            }
        }
 
        return false;
    }
}
