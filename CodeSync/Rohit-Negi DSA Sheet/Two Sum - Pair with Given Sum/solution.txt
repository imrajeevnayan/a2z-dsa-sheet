import java.util.HashSet;

class Solution {
    boolean twoSum(int arr[], int target) {
        HashSet<Integer> seen = new HashSet<>();

        for (int num : arr) {
            int complement = target - num;

            // Agar complement pehle se set mein hai, toh pair mil gaya
            if (seen.contains(complement)) {
                return true;
            }

            // Nahi toh current number ko set mein add karo
            seen.add(num);
        }

        return false;
    }
}
