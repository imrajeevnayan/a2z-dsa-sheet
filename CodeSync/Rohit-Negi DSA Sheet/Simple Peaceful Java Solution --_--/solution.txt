class Solution {
    public Node removeDuplicates(Node head) {
        HashSet<Integer>set=new HashSet<>();
        Node curr=head,prev=null;
        while(curr!=null){
            if(set.contains(curr.data)){
                // duplicate ko remove kro
                prev.next=curr.next;
            }
            else{
                // valu
                set.add(curr.data);
                prev=curr;
            }
            curr=curr.next;
        }
       return head; 
    }
}
