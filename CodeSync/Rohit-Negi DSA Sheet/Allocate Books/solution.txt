import java.util.*;
 
public class Solution {
 
    public static long ayushGivesNinjatest(int n, int m, int[] time) {
 
        // Minimum answer = sabse bada chapter
        // Kyunki ek chapter ko tod nahi sakte
        long low = 0;
 
        // Maximum answer = saare chapters ek hi din mein
        long high = 0;
 
        for (int i = 0; i < m; i++) {
            low = Math.max(low, time[i]);
            high += time[i];
        }
 
        long ans = high;
 
        // Binary Search on Answer
        while (low <= high) {
 
            long mid = low + (high - low) / 2;
 
            // Check karenge ki mid maximum daily time ke andar
            // saare chapters n days mein complete ho sakte hain ya nahi
            if (isPossible(time, m, n, mid)) {
 
                // Possible hai, to aur chhota answer try karo
                ans = mid;
                high = mid - 1;
 
            } else {
 
                // Possible nahi hai, daily limit badhani padegi
                low = mid + 1;
            }
        }
 
        return ans;
    }
