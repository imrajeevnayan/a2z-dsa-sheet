class Solution {
    public int findTheDistanceValue(int[] arr1, int[] arr2, int d) {
        Arrays.sort(arr2);
        int count = 0;
        for (int num : arr1) {
            if (isValid(arr2, num, d))  count++;
        }
        return count;
    }
    private boolean isValid(int[] arr2, int target, int d) {

        while (left <= right) {
            int mid = left + (right - left) / 2;
