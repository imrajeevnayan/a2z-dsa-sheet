class Solution {
    public boolean canAliceWin(int n) {
        int remove = 10;
        boolean aliceTurn = true;

        while (n >= remove) {
            n -= remove;
            remove--;
            aliceTurn = !aliceTurn;
        }

        // aliceTurn indicates whose turn it is now.
        // If it's Alice's turn and she can't move -> Alice loses.
        // Otherwise Bob can't move -> Alice wins.
        return !aliceTurn;
    }
}
