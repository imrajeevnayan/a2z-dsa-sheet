class Solution {
    public int findMissing(int[] arr) {

        int n = arr.length;

        int d = (arr[n - 1] - arr[0]) / n;

        for (int i = 1; i < n; i++) {
            if (arr[i] - arr[i - 1] != d)
                return arr[i - 1] + d;
        }

        return arr[0] + d; 
    }
}
