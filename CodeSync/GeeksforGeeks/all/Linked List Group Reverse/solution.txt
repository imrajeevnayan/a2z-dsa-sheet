class Solution {
    public Node reverseKGroup(Node head, int k) {
        if (head == null || k == 1) return head;
        

        Node curr = head;
        Node prev = null;
        Node next = null;
        int count = 0;

        while (curr != null && count < k) {
            next = curr.next;
            curr.next = prev;
            prev = curr;
            curr = next;
            count++;
        }
        head.next = reverseKGroup(curr, k);

        return prev;
    }
}
