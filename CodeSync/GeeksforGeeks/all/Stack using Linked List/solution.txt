// Stack class
class myStack {
    private Node head; // Head acts as the Top of the stack
    private int count; // To track size in O(1)

    public myStack() {
        // Initialize your data members
        this.head = null;
        this.count = 0;
    }

    public boolean isEmpty() {
        // check if the stack is empty
        return head == null;
    }

    public void push(int x) {
        // Adds an element x at the top of the stack.
        Node newNode = new Node(x);
        newNode.next = head; // New node points to current top
        head = newNode;      // Update top to be the new node
        count++;             // Increment size
    }

    public void pop() {
        // Removes the top element of the stack.
        if (isEmpty()) {
            return; // Do nothing if empty
        }
        head = head.next; // Move top pointer to next node
        count--;          // Decrement size
    }

    public int peek() {
        // Returns the top element of the stack.
        // If stack is empty, return -1.
        if (isEmpty()) {
            return -1;
        }
        return head.data;
    }
