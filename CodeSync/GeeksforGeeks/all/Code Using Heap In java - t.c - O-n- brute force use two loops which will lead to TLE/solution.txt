class Solution {
    public static int kthLargest(int arr[], int k) {
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        for (int x : arr) pq.add(x);
        
        for (int i = 0; i < arr.length - k; i++) pq.poll();

        return pq.poll();
    }
}
