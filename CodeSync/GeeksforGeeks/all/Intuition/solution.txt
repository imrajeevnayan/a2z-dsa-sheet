class Solution {
    static ArrayList<Integer> nextSmallerEle(int[] arr) {
        int n = arr.length;
        ArrayList<Integer> result =  new ArrayList<>(Collections.nCopies(n, -1));
        Deque<Integer> st = new ArrayDeque<>();

        for (int i = n-1; i >=0; i--) {
            while (!st.isEmpty() && arr[i] > arr[st.peek()]) {
                result.set(st.pop(), arr[i]);
            }
            st.push(i);
        }
        return result;
    }
}
