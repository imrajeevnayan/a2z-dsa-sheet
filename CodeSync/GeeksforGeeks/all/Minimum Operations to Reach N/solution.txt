class Solution {
    public int minOperation(int n) {
        if (n == 0) return 0;

            // Option 1: Last op was +1 (from n-1)
            int opt1 = 1 + minOperation(n - 1);

            // Option 2: Last op was *2 (only if even)
            int opt2 = Integer.MAX_VALUE;
            if (n % 2 == 0) {
                opt2 = 1 + minOperation(n / 2);
            }

            return Math.min(opt1, opt2);
    }
}
