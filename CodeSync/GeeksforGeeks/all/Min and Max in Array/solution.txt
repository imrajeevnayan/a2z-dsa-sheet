class Solution {
    public ArrayList<Integer> getMinMax(int[] arr) {
      int min=Integer.MAX_VALUE;
      int max=Integer.MIN_VALUE;
      for(int n :arr){
          if(n < min) min=n;
          if(n > max) max=n;
      }
       return new ArrayList<>(Arrays.asList(min,max));
    }
}
