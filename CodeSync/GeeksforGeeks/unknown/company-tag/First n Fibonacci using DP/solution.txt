class Solution {
    public ArrayList<Integer> fibonacciNumbers(int n) {
        ArrayList<Integer> fib = new ArrayList<>();
        int MOD = 1000000007;
        
        long[] dp = new long[n + 1];
        
        dp[0] = 0;
        if (n >= 1) dp[1] = 1;
        
       for (int i = 2; i <= n; i++) {
            dp[i] = (dp[i - 1] + dp[i - 2]) % MOD;
        }
        
        for (int i = 0; i <= n; i++) {
            fib.add((int) dp[i]);
        }
        
        return fib;
    }
}
