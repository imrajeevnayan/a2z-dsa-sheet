    void merge(int[] arr, int[] idx, int[] ans, int l, int m, int r) {
        int[] temp = new int[r - l + 1];
        int i = l, j = m + 1, k = 0, count = 0;

        while (i <= m && j <= r) {
            if (arr[idx[j]] < arr[idx[i]]) {
                count++;
                temp[k++] = idx[j++];
            } else {
                ans[idx[i]] += count;
                temp[k++] = idx[i++];
            }
        }

        while (i <= m) {
            ans[idx[i]] += count;
            temp[k++] = idx[i++];
        }

        while (j <= r)
            temp[k++] = idx[j++];

        for (i = l; i <= r; i++)
            idx[i] = temp[i - l];
    }
