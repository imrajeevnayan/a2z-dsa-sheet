class Solution {
    public ArrayList<Integer> nextLargerElement(int[] arr) {
        int n = arr.length;
        ArrayList<Integer> result = new ArrayList<>(Collections.nCopies(n, -1));
        Deque<Integer> st = new ArrayDeque<>();

        for (int i = 0; i <= n; i++) {
            while (!st.isEmpty() && arr[i]>arr[st.peek()] ) {
                result.set()
            }
    
            st.push(arr[i]);
        }
        return result;
    }
}
