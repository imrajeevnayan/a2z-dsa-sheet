class Solution {
    public ArrayList<Integer> leftView(Node root) {
        ArrayList<Integer>ans=new ArrayList<>();
        if (root == null) return ans;

           Queue<TreeNode> q = new LinkedList<>();
           q.offer(root);

           while (!q.isEmpty()) {

               int size = q.size();

               for (int i = 0; i < size; i++) {

                   TreeNode node = q.poll();

                   if (i == 0) {
                       ans.add(node.val);
                   }

                   if (node.left != null) {
                       q.offer(node.left);
                   }

                   if (node.right != null) {
                       q.offer(node.right);
                   }
               }
           }

           return ans; 
    }
}
