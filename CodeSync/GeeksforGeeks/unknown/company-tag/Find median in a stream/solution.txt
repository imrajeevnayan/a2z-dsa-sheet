import java.util.ArrayList;
import java.util.PriorityQueue;

class Solution {
    public ArrayList<Double> getMedian(int[] arr) {
        PriorityQueue<Integer> left = new PriorityQueue<>((a, b) -> b - a); // Max-heap
        PriorityQueue<Integer> right = new PriorityQueue<>();               // Min-heap
        ArrayList<Double> result = new ArrayList<>();

        for (int num : arr) {
            // Add to appropriate heap
            if (left.isEmpty() || num <= left.peek()) {
                left.offer(num);
            } else {
                right.offer(num);
            }

            // Balance heaps: size difference should be at most 1
            if (left.size() > right.size() + 1) {
                right.offer(left.poll());
            } else if (right.size() > left.size()) {
                left.offer(right.poll());
            }

            // Calculate current median
            double median;
            if (left.size() == right.size()) {
                median = (left.peek() + right.peek()) / 2.0;
            } else {
                median = left.peek();
            }
            result.add(median);
        }

        return result;
