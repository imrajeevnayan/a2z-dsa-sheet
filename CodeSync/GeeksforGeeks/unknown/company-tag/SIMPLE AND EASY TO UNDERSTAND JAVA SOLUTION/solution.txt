class Solution {
    int[] dp;

    public int[] countBits(int n) {
        dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;

        for (int i = 0; i <= n; i++)
            solve(i);

        return dp;
    }

    int solve(int n) {
        if (dp[n] != -1)
            return dp[n];

        return dp[n] = solve(n / 2) + n % 2;
    }
}
