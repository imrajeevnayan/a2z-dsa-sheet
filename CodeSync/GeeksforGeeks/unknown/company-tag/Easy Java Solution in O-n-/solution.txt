class Solution {
    public int minParentheses(String s) {

        int open = 0;
        int count = 0;

        for (char ch : s.toCharArray()) {

            if (ch == '(') {
                open++;
            } 
            else {
                if (open > 0) {
                    open--;
                } 
                else {
                    count++;
                }
            }
        }

        // Jo '(' bach gaye, unke liye ')' add karne padenge
        count += open;

        return count;
    }
}
