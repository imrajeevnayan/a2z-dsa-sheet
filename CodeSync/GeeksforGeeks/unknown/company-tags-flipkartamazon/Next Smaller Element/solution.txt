class Solution {
    static ArrayList<Integer> nextSmallerEle(int[] arr) {
        int n=arr.length;
        ArrayList<Integer>ans=new ArrayList<>(Collections.nCopies(n,-1));
        Stack<Integer>st=new Stack<>();
        for(int i=0;i<n;i++){
            while(!st.isEmpty()&& arr[i]< arr[st.peek()]){
                ans.set(st.peek(),arr[i]);
                st.pop();
            }
            st.push(i);
        }
        return ans;
    }
}
