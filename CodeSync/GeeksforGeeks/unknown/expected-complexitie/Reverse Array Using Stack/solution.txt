class Solution {
    public void reverseArray(int[] arr) {
        Stack<Integer> st = new Stack<>();
        for (int num : arr)st.push(num);
        for (int i = 0; i < arr.length; i++) {
            arr[i] = st.pop();
        }
    }
}
