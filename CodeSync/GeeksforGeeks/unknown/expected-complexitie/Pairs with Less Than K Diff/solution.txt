class Solution {
    public static int countPairs(int arr[], int k) {
        Arrays.sort(arr);
        int left = 0,count = 0;
        for (int right = 1; right < arr.length; right++) {
            while (arr[right] - arr[left] >= k) {
                left++;
            }
            count += right - left;
        }
        return count;
    }
}
