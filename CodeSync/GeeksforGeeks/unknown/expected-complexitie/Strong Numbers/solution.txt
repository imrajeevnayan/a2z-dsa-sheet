class Solution {
    public boolean isStrong(int n) {
        int[] fact = {1, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880};

        int original = n;
        int sum = 0;

        while (n > 0) {
            sum += fact[n % 10];
            n /= 10;
        }

        return sum == original;
    }
}
