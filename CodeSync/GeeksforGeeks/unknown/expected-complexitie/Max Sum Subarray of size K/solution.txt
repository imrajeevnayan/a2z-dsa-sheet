class Solution {
    public int maxSubarraySum(int[] arr, int k) {
        int left = 0,currSum = 0,ans = Integer.MIN_VALUE;
          for (int right = 0; right < arr.length; right++) {
                    // 1. Right element ko window mein add karo
                    currSum += arr[right];

                    // 2. Window size == k
                    if (right - left + 1 == k) {
                        // 3. Answer calculate/update karo
                        ans = Math.max(ans, currSum);
                        // 4. Left element ko window se remove karo
                        currSum -= arr[left];
                        left++;
                  }
                }
                return ans;
            }
        }
