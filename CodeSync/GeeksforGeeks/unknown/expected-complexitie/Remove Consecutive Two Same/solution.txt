class Solution {
    public String removePair(String s) {
        Stack<Character>st=new Stack<>();
        StringBuilder sb=new StringBuilder();
        for(char c:s.toCharArray()){
            if(!st.isEmpty()&& st.peek()==c)st.pop();
            else st.push(c);
        }
        if(st.isEmpty())return "-1";
        while(!st.isEmpty()){
            sb.append(st.pop());
        }
        return sb.reverse().toString();
        
    }
}
