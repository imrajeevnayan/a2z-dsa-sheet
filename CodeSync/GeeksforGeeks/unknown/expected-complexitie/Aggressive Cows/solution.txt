class Solution {
    class Solution {

        public boolean canPlace(int[] arr, int k, int distance) {

            int count = 1;
            int lastPosition = arr[0];

            for (int i = 1; i < arr.length; i++) {

                if (arr[i] - lastPosition >= distance) {
                    count++;
                    lastPosition = arr[i];
                }

                if (count == k) {
                    return true;
                }
            }

            return false;
        }
