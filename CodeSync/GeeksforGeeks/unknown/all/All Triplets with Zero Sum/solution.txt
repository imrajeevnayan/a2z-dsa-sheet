class Solution {
    public List<List<Integer>> findTriplets(int[] arr) {
        List<List<Integer>> result = new ArrayList<>();
        int n = arr.length;

        for (int i = 0; i < n - 2; i++) {
            HashMap<Integer, Integer> map = new HashMap<>();

            for (int j = i + 1; j < n; j++) {
                int required = -(arr[i] + arr[j]);

                if (map.containsKey(required)) {
                    result.add(Arrays.asList(
                        arr[i],
                        required,
                        arr[j]
                    ));
                }

                map.put(arr[j], j);
            }
        }

        return result;
    }
}
