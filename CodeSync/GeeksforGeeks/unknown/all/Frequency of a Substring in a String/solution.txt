class Solution {
    public int countFreq(String s1, String s2) {
        int n = s1.length();
        int m = s2.length();

        int[] lps = new int[m];
        buildLPS(s2, lps);

        int i = 0, j = 0;
        int count = 0;

        while (i < n) {
            if (s1.charAt(i) == s2.charAt(j)) {
                i++;
                j++;

                if (j == m) {
                    count++;
                    j = lps[j - 1]; // for overlapping matches
                }
            } else {
                if (j != 0)
                    j = lps[j - 1];
                else
                    i++;
            }
        }

        return count;
    }
