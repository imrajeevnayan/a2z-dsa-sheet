class Solution {
    public List<Integer> findSplit(int[] arr) {
        List<Integer> ans = new ArrayList<>();
        long total = 0;

        for (int x : arr) {
            total += x;
        }

        if (total % 3 != 0) {
            ans.add(-1);
            ans.add(-1);
            return ans;
        }

        long target = total / 3;
        long sum = 0;
        int first = -1;

        for (int i = 0; i < arr.length - 1; i++) {
            sum += arr[i];

            if (sum == target && first == -1) {
                first = i;
            }

            if (sum == 2 * target && first != -1) {
                ans.add(first);
                ans.add(i);
                return ans;
            }
        }

        ans.add(-1);
        ans.add(-1);
        return ans;
    }
}
