class Complete {
    public static int[] Smallestonleft(int arr[], int n) {
        int[] ans = new int[n];
        int min = arr[0];
        ans[0] = -1;
        for (int i = 1; i < n; i++) {
            if (min < arr[i])  ans[i] = min;
            else  ans[i] = -1;

            min = Math.min(min, arr[i]);
        }

        return ans;
    }
}
