class Solution {
    boolean areElementsContiguous(int arr[]) {
        HashMap<Integer, Integer> map = new HashMap<>();

        int min = arr[0];
        int max = arr[0];

        for (int num : arr) {
            map.put(num, map.getOrDefault(num, 0) + 1);

            min = Math.min(min, num);
            max = Math.max(max, num);
        }

        // Number of distinct elements must equal range size
        if (map.size() != max - min + 1) {
            return false;
        }

        // Check every value in the range exists
        for (int i = min; i <= max; i++) {
            if (!map.containsKey(i)) {
                return false;
            }
        }

        return true;
    }
}
