class Solution {
    static ArrayList<Long> factorialNumbers(long n) {
        ArrayList<Long> ans = new ArrayList<>();
        helper(1, 1, n, ans);
        return ans;
    }

    static void helper(long fact, long i, long n, ArrayList<Long> ans) {
        fact = fact * i;
        if (fact > n)  return;
        ans.add(fact);
        helper(fact, i + 1, n, ans);
    }
}
