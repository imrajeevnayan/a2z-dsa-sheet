class Solution {
     private int solve(int num, int[] dp) {
        if (num == 0) return 0;
        if (dp[num] != -1)  return dp[num];

        // Choice 1: Pichla step "+1" tha (Source: num - 1)
        int addOne = 1 + solve(num - 1, dp);

        // Choice 2: Pichla step "*2" tha (Source: num / 2)
        // Ye tabhi possible hai jab num EVEN ho
        int doubleOp = Integer.MAX_VALUE;
        if (num % 2 == 0) {
            doubleOp = 1 + solve(num / 2, dp);
        }

        // Result: Dono choices mein se MINIMUM lo
        dp[num] = Math.min(addOne, doubleOp);

        return dp[num];
    }
    public int minOperation(int n) {
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0; 
        return solve(n, dp);
    }
}
