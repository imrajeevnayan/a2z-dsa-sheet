class Solution {
    public int minValueToBalance(List<Integer> arr) {
        int mid = arr.size() / 2;
        int left = 0, right = 0;

        for (int i = 0; i < mid; i++)
            left += arr.get(i);

        for (int i = mid; i < arr.size(); i++)
            right += arr.get(i);

        return Math.abs(left - right);
    }
}
