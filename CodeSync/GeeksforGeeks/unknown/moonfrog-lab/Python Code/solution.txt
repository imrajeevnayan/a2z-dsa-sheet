class Solution {
    int solve(int i, int[] arr) {
        int n = arr.length;
        if (i >= n - 1) return 0;
        if (arr[i] == 0) return (int)1e9;
        int ans = (int)1e9;
        for (int jump = 1; jump <= arr[i] && i + jump < n; jump++) {
            ans = Math.min(ans, 1 + solve(i + jump, arr));
        }
        return ans;
    }
    public int minJumps(int[] arr) {
        int ans = solve(0, arr);
        return ans >= (int)1e9 ? -1 : ans;
    }
}
