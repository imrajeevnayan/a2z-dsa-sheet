            maxi = Math.max(maxi, num);

        // If elements are consecutive, then 
        // there should be maxi-mini+1 elements.
        if (maxi - mini + 1 != n) return false;
        
        boolean[] visited = new boolean[n];
        
        for (int num : arr) {
            
            // If element is already marked
            if (visited[num - mini]) {
                return false;
            }
            
            // Mark the element.
            visited[num - mini] = true;
        }
        
        return true;
    }

        for (int num : arr) {
            mini = Math.min(mini, num);
            maxi = Math.max(maxi, num);
        }
        
        // If elements are consecutive, then

        if (maxi - mini + 1 != n) return false;
        
        for (int i = 0; i < n; i++) {
            
            int val = Math.abs(arr[i]);
            
            // If value is already marked
            // as negative, return false 
            // as it is already present.
            if (arr[val - mini] < 0) {
                return false;
            }
            
            // Mark the value

        }
        
        return true;
