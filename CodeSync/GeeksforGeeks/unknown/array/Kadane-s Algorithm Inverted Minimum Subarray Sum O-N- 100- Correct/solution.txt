class Solution {
    static int smallestSumSubarray(int[] a, int size) {
        int cur = a[0], ans = a[0];
        for (int i = 1; i < size; i++) {
            cur = Math.min(a[i], cur + a[i]);
            ans = Math.min(ans, cur);
        }
        return ans;
    }
}
