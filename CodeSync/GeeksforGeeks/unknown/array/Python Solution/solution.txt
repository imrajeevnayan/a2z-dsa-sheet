class Solution {
    public List<List<Integer>> distinctPairs(int[] arr, int target) {
        List<List<Integer>> ans = new ArrayList<>();
        Arrays.sort(arr);

        int l = 0, r = arr.length - 1;

        while (l < r) {
            int sum = arr[l] + arr[r];

            if (sum == target) {
                ans.add(Arrays.asList(arr[l], arr[r]));

                int a = arr[l], b = arr[r];
                while (l < r && arr[l] == a) l++;
                while (l < r && arr[r] == b) r--;
            } 
            else if (sum < target) l++;
            else r--;
        }

        return ans;
    }
}
