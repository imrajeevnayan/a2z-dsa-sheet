class Solution {
    public boolean areConsecutives(int[] arr) {
        Arrays.sort(arr);
        for (int i = 1; i < arr.length; i++) {
            if (arr[i] == arr[i - 1])  return false;

            if (arr[i] != arr[i - 1] + 1) return false;
        }
        return true;
    }
}
