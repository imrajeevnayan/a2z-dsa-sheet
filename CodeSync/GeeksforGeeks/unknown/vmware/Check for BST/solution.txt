class Solution {
    Node prev = null;

    public boolean isBST(Node root) {
        if (root == null)  return true;

        if (!isBST(root.left)) return false;
        if (prev != null && root.data <= prev.data)  return false;

        prev = root;
        return isBST(root.right);
    }
}
