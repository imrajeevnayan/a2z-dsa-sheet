        // Empty list
        if (head == null) {
            return null;
        }
 
        // Only one node
        if (head.next == head) {
            if (head.data == key) {
                return null;
            }
            return head;
        }
 
        // If head needs to be deleted
        if (head.data == key) {
            Node<Integer> temp = head;
 
            // Find the last node
            while (temp.next != head) {
                temp = temp.next;
            }
 
            // Last node points to second node
            temp.next = head.next;
 
            return head.next;
