class Solution {
    public int minJumps(int[] arr) {
        int n = arr.length;
        Integer[] dp = new Integer[n];
        return solve(0, arr, dp);
    }
    
    private int solve(int i, int[] arr, Integer[] dp) {
        if (i >= arr.length - 1) return 0;
        if (dp[i] != null) return dp[i];
        
        int ans = Integer.MAX_VALUE;
        int maxJump = arr[i];
        
        for (int j = 1; j <= maxJump && i + j < arr.length; j++) {
            int jumps = solve(i + j, arr, dp);
            if (jumps != Integer.MAX_VALUE) {
                ans = Math.min(ans, 1 + jumps);
            }
        }
        
        return dp[i] = (ans == Integer.MAX_VALUE ? Integer.MAX_VALUE : ans);
    }
}
