/*
3 approach se solve hoga:
1. Reverse the linked list
2. Add 1 to linked list
3. Reverse the linked list
*/

class Solution {

    // Reverse linked list
    private Node reverse(Node head) {
        Node prev = null;
        Node curr = head;

        while (curr != null) {
            Node temp = curr.next;
            curr.next = prev;
            prev = curr;
            curr = temp;
        }

        return prev;
    }

    public Node addOne(Node head) {
        // Step 1: Reverse the linked list
        head = reverse(head);
