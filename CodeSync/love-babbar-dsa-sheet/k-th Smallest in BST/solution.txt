class Solution {
    public int kthSmallest(Node root, int k) {
        PriorityQueue<Integer> maxHeap = new PriorityQueue<>(Collections.reverseOrder());
        inorderTraversal(root, maxHeap, k);
        return maxHeap.peek();
    }

    private void inorderTraversal(Node node, PriorityQueue<Integer> maxHeap, int k) {
        if (node == null) {
            return;
        }
        inorderTraversal(node.left, maxHeap, k);
        maxHeap.add(node.data);
        
        if (maxHeap.size() > k) {
            maxHeap.poll();
        }
        inorderTraversal(node.right, maxHeap, k);
    }
}
