class Solution {
    public int maxLength(String s) {
        int maxLen = 0;
        int open = 0, close = 0;

        // --- Pass 1: Left to Right ---
        // Ye handle karta hai cases jahan ')' zyada ho jayein (e.g., ")()")
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == '(') {
                open++;
            } else {
                close++;
            }

            // Agar balance match ho gaya
            if (open == close) {
                maxLen = Math.max(maxLen, 2 * open);
            } 
            // Agar closing brackets zyada ho gaye, toh sequence invalid hai -> RESET
            else if (close > open) {
                open = 0;
                close = 0;
            }
        }

        // Reset variables for second pass
        open = 0;
        close = 0;

                close++;
