class Solution {
    public static int smallestSubWithSum(int x, int[] arr) {
        int n = arr.length;
        int left = 0;
        int currSum = 0;
        int minLen = Integer.MAX_VALUE;

        // Right pointer 'i' for loop se control hoga
        for (int right = 0; right < n; right++) {
            // 1. Expand: Current element add karo
            currSum += arr[right];

            // 2. Shrink: Jab tak sum > x hai, left pointer ko badhao
            while (currSum > x) {
                // Update min length
                minLen = Math.min(minLen, right - left + 1);

                // Left element remove karo aur left badhao
                currSum -= arr[left];
                left++;
            }
        }

        return (minLen == Integer.MAX_VALUE) ? 0 : minLen;
    }
}
