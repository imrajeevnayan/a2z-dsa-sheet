int price;
        int stocks;

        Pair(int price, int stocks) {
            this.price = price;
            this.stocks = stocks;
        }
    }

    public static int buyMaximumProducts(int k, int[] price) {

        PriorityQueue<Pair> pq =
                new PriorityQueue<>((a, b) -> a.price - b.price);

        // Heap me sab days daal do
        for (int i = 0; i < price.length; i++) {
            pq.offer(new Pair(price[i], i + 1));
        }

        int ans = 0;

        while (!pq.isEmpty()) {

            Pair curr = pq.poll();

            int cost = curr.price * curr.stocks;

            // Agar poore stocks kharid sakte hain
            if (k >= cost) {
                k -= cost;
                ans += curr.stocks;
            } else {

                // Jitne afford kar sakte hain
                ans += k / curr.price;
                break;
            }
        }

        return ans;
    }
