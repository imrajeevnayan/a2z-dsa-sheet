import java.util.List;

public class Solution {
 
    static int findKthAncestor(BinaryTreeNode<Integer> root,
                               int targetNodeVal, int kth) {
 
        List<BinaryTreeNode<Integer>> path = new ArrayList<>();
 
        if (!findPath(root, targetNodeVal, path)) {
            return -1;
        }
 
        int index = path.size() - 1 - kth;
 
        if (index < 0) {
            return -1;
        }
 
        return path.get(index).data;
    }
