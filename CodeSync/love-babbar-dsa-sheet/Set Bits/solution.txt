import java.util.* ;
import java.io.*; 
public class Solution {
  public static int countSetBits(int n) {
    // Write your code here.
    int count = 0;
 
        while (n > 0) {
        count += n & 1;
        n = n >> 1;
    }
 
      return count;
  }
}
