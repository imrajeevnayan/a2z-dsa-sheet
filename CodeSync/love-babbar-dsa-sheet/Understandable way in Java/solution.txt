class Solution {
    public int secFrequent(String[] arr) {
        HashMap<String, Integer> map = new HashMap<>();

        for (String key : arr) {
            map.put(key, map.getOrDefault(key, 0) + 1);
        }

        int first = 0;
        int second = 0;

        for (String key : map.keySet()) {
            int freq = map.get(key);

            if (freq > first) {
                second = first;
                first = freq;
            } else if (freq > second && freq < first) {
                second = freq;
            }
        }

        return second;
    }
}
