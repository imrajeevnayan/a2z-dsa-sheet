class Solution {
    int multiplyTwoLists(Node first, Node second) {
        long mod = 1000000007;
        long num1 = 0;
        long num2 = 0;

        while (first != null) {
            num1 = (num1 * 10 + first.data) % mod;
            first = first.next;
        }

        while (second != null) {
            num2 = (num2 * 10 + second.data) % mod;
            second = second.next;
        }

        return (int)((num1 * num2) % mod);
    }
}
