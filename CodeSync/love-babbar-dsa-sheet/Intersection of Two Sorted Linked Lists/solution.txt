/* Structure of a Linked list Node
 class Node {
    int data;
    Node next;

    Node(int x) {
        data = x;
        next = null;
    }
} */

class Solution {
    public Node findIntersection(Node head1, Node head2) {
        Set<Node>set=new HashSet<>();
        while(head1 !=null){
            set.add(head1);
            head1=head1.next;
        }
        while(head2 !=null){
            if(set.contains(head1)) return head1;
            head2=head2.next;
        }
        return null;
        
    }
}
