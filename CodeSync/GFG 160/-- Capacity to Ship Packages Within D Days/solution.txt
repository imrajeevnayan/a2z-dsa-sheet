class Solution {
    public int leastWeightCapacity(ArrayList<Integer> arr, int d) {

        int low = 0;
        int high = 0;
        int ans = 0;

        for (int x : arr) {
            low = Math.max(low, x);
            high += x;
        }

        while (low <= high) {

            int mid = low + (high - low) / 2;

            int days = 1;
            int sum = 0;

            for (int x : arr) {

                if (sum + x > mid) {
                     days++;
                    sum = 0;
                }

                sum += x;
            }

            if (days <= d) {
                ans = mid;
                high = mid - 1;
            } 
            else {
                low = mid + 1;
            }
        }

        return ans;
    }
}
