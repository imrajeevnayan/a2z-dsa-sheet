class Solution {
    public ArrayList<Integer> asciirange(String s) {
        int n = s.length();

        int[] first = new int[26];
        int[] last = new int[26];

        Arrays.fill(first, -1);
        Arrays.fill(last, -1);

        for (int i = 0; i < n; i++) {
            int idx = s.charAt(i) - 'a';

            if (first[idx] == -1) {
                first[idx] = i;
            }
            last[idx] = i;
        }

        long[] prefix = new long[n + 1];

        for (int i = 0; i < n; i++) {
            prefix[i + 1] = prefix[i] + s.charAt(i);
        }

        ArrayList<Integer> ans = new ArrayList<>();

        for (int i = 0; i < 26; i++) {
            if (first[i] != -1 && first[i] < last[i]) {

                long sum = prefix[last[i]] - prefix[first[i] + 1];

                if (sum != 0) {
                    ans.add((int) sum);
                }
            }
        }

        return ans;
    }
}
