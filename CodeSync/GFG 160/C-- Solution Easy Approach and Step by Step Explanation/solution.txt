class Solution {
    public int countAlwaysFound(int[] arr) {
        int n = arr.length;
        int[] sorted = arr.clone();
        Arrays.sort(sorted);

        int count = 0;

        for (int i = 0; i < n; i++) {
            if (arr[i] == sorted[i]) {
                count++;
            }
        }

        return count;
    }
}
