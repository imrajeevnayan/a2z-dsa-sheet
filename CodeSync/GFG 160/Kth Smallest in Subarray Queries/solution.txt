    int upperBound(ArrayList<Integer> list, int x) {
        int lo = 0, hi = list.size();

        while (lo < hi) {
            int mid = (lo + hi) / 2;

            if (list.get(mid) <= x)
                lo = mid + 1;
            else
                hi = mid;
        }

        return lo;
    }
