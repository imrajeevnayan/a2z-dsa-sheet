class Solution {
    void segregate0and1(int[] arr) {
     int j=0;
     for(int i=0;i<arr.length;i++){
         if(arr[i]==0) arr[j++]=0;
     }
     while(j<arr.length)arr[j++]=1;
        
    }
}
