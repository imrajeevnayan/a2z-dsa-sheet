class Solution {
    public int minPlatform(int arr[], int dep[]) {
        Arrays.sort(arr);
        Arrays.sort(dep);
        
        int n = arr.length;
        int platforms = 0, maxPlatforms = 0;
        int i = 0, j = 0; 
        while (i < n && j < n) {
            if (arr[i] <= dep[j]) {
                platforms++;      
                i++;
            } else {
                platforms--;
                j++;
            }
            maxPlatforms = Math.max(maxPlatforms, platforms);
        }
        return maxPlatforms;
    }
}
