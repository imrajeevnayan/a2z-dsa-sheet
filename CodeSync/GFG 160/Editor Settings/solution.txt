class Solution {
    int countPairs(int[] arr, int k) {
        HashMap<Integer,Integer>map=new HashMap <>();
        for(int  n :arr){
            map.put(n,map.getOrDefault(n,0)+1);
        }
        long count=0;
        for(int n :arr){
            count+=map.getOrDefault(n+k,0);
        }
        return (int)count;
    }
}
