class Solution {
    public String findNth(String n) {

        long num = Long.parseLong(n);

        StringBuilder sb = new StringBuilder();

        while (num > 0) {
            sb.append(num % 9);
            num /= 9;
        }

        return sb.reverse().toString();
    }
}
