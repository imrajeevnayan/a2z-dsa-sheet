class Solution {
    public int minJumps(int[] arr) {
        int n = arr.length;
        if (n <= 1) return 0;
        if (arr[0] == 0) return -1;
        
        int jumps = 0;       // Number of jumps taken
        int currEnd = 0;     // Farthest index reachable with current number of jumps
        int farthest = 0;    // Farthest index reachable overall so far
        
        for (int i = 0; i < n - 1; i++) {
            farthest = Math.max(farthest, i + arr[i]);
            
            // If we've reached the end of the current jump's range
            if (i == currEnd) {
                jumps++;
                currEnd = farthest;
                
                // If we can already reach or pass the last index
                if (currEnd >= n - 1) {
                    return jumps;
                }
            }
        }
        
        return -1; // Should not reach here if input is valid
    }
}
