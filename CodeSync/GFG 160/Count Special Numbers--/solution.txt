int max = 0;
        for (int x : arr) {
            max = Math.max(max, x);
        }

        int[] freq = new int[max + 1];
        for (int x : arr) {
            freq[x]++;
        }

        int[] divisors = new int[max + 1];

        // Count how many array elements divide each number
        for (int d = 1; d <= max; d++) {
            if (freq[d] > 0) {
                for (int multiple = d; multiple <= max; multiple += d) {
                    divisors[multiple] += freq[d];
                }
            }
        }

        int ans = 0;

        for (int val = 1; val <= max; val++) {

            if (freq[val] == 0)
                continue;

            if (freq[val] > 1) {
                // same-valued elements divide each other
                ans += freq[val];
            } else {
                // divisor count > 1 means some other divisor exists
                if (divisors[val] > 1) {
                    ans++;
                }
            }
        }

        return ans;
    }
