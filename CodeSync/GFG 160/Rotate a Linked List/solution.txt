class Solution {
    public Node rotate(Node head, int k) {
        if (head == null || head.next == null || k == 0) {
            return head;
        }

        // Find length
        int n = 1;
        Node tail = head;

        while (tail.next != null) {
            tail = tail.next;
            n++;
        }

        k = k % n;

        if (k == 0) {
            return head;
        }

        // Find kth node
        Node curr = head;
        for (int i = 1; i < k; i++) {
            curr = curr.next;
        }

        // New head is k+1 node
        Node newHead = curr.next;

        // Connect old tail to old head
        tail.next = head;

        // Break at kth node
        curr.next = null;

        return newHead;
