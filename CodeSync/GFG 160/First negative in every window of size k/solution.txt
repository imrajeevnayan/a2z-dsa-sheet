class Solution {
    static List<Integer> firstNegInt(int arr[], int k) {
        List<Integer> ans = new ArrayList<>();
        Queue<Integer> q = new LinkedList<>();
        // First window
        for (int i = 0; i < k; i++) {
            if (arr[i] < 0) q.offer(arr[i]);
        }
        ans.add(q.isEmpty() ? 0 : q.peek());

        // Remaining windows
        for (int i = k; i < arr.length; i++) {

            // Remove outgoing negative
            if (!q.isEmpty() && arr[i - k] == q.peek()) {
                q.poll();
            }

            // Add incoming negative
            if (arr[i] < 0) {
                q.offer(arr[i]);
            }

            ans.add(q.isEmpty() ? 0 : q.peek());
        }

        return ans;
    }
}
