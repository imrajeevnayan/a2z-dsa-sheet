import java.util.*;

class Solution {
    public int countPairs(int[] arr, int k) {
        HashMap<Integer, Integer> map = new HashMap<>();
        long count = 0;

        for (int num : arr) {
            int rem = num % k;

            count += map.getOrDefault(rem, 0);

            map.put(rem, map.getOrDefault(rem, 0) + 1);
        }

        return (int) count;
    }
}
