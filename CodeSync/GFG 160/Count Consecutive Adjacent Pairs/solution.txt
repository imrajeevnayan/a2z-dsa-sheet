class Solution {
    public int adjacentPairs(int[] arr) {

        int count = 0;

        // Last element tak nahi jana,
        // kyunki hume i aur i+1 compare karna hai
        for (int i = 0; i < arr.length - 1; i++) {

            // Check karo next element current se exactly 1 bada hai ya nahi
            if (arr[i + 1] == arr[i] + 1) {
                count++;
            }
        }

        return count;
    }
}
