class Solution {
    int countPairs(int arr[], int target) {
        HashMap<Integer, Integer> map = new HashMap<>();
        int count = 0;

        for (int num : arr) {
            int rem = target - num;
            count += map.getOrDefault(rem, 0);
            map.put(num, map.getOrDefault(num, 0) + 1);
        }
        return count;
    }
}
