class Solution {
    // Cheat sheet to store already calculated results
    int[] memo; 

    int getMinSteps(int n) {
        // Initialize memo array only once (or if null)
        if (memo == null || memo.length <= n) {
            memo = new int[n + 1];
        }

        // Base Case
        if (n <= 1) return 0;

        // Step 1: Check if answer is already in cheat sheet
        if (memo[n] != 0) return memo[n];

        // Step 2: Pure Recursion Logic (Same as before)
        int res = getMinSteps(n - 1) + 1;

        if (n % 2 == 0) {
            res = Math.min(res, getMinSteps(n / 2) + 1);
        }

        if (n % 3 == 0) {
            res = Math.min(res, getMinSteps(n / 3) + 1);
        }

        // Step 3: Save result to cheat sheet before returning
        memo[n] = res;

        return res;
    }
}
