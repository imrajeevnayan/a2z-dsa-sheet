class Solution {

    class Pair {
        int value;
        int diff;

        Pair(int value, int diff) {
            this.value = value;
            this.diff = diff;
        }
    }

    public void rearrange(int[] arr, int x) {

        List<Pair> list = new ArrayList<>();

        // value aur uska absolute difference store karo
        for (int num : arr) {
            list.add(new Pair(num, Math.abs(num - x)));
        }

        // Difference ke basis par stable sort
        Collections.sort(list, (a, b) -> a.diff - b.diff);

        // Wapas original array me bhar do
        for (int i = 0; i < arr.length; i++) {
            arr[i] = list.get(i).value;
        }
    }
}
