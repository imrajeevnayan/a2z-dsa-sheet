import java.util.*;
import java.io.*;
 
public class Solution {
    static class MinStack {
        Stack<Integer> stack;
        Stack<Integer> minStack;
        MinStack() {
            stack = new Stack<>();
            minStack = new Stack<>();
        }
        void push(int num) {
            stack.push(num);
            if (minStack.isEmpty() || num <= minStack.peek()) {
                minStack.push(num);
            }
        }
        int pop() {
            if (stack.isEmpty())
                return -1;
 
            int x = stack.pop();
 
            if (x == minStack.peek()) {
                minStack.pop();
            }
 
            return x;
        }
