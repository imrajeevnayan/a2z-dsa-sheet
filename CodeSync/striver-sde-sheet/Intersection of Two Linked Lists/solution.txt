public class Solution {
    public static int findIntersection(Node firstHead, Node secondHead) {
        Node a = firstHead,b = secondHead;
 
        while (a != b) {
            a = (a == null) ? secondHead : a.next;
            b = (b == null) ? firstHead : b.next;
        }
 
        return a == null ? -1 : a.data;
    }
}
