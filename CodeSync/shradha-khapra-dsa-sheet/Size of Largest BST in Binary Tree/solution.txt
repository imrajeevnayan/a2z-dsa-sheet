class NodeInfo {
    int min;
    int max;
    int size;
 
    NodeInfo(int min, int max, int size) {
        this.min = min;
        this.max = max;
        this.size = size;
    }
}

    public static int largestBST(TreeNode root) {
        return solve(root).size;
    }

                                0);
