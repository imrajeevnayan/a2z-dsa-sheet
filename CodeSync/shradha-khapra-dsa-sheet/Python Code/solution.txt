class Solution {
    int solve(int i, int[] arr, int[] dp) {
        int n = arr.length;
        if (i >= n - 1) return 0;
        if (arr[i] == 0) return (int)1e9;
        if (dp[i] != -1)  return dp[i];
        int ans = (int)1e9;
        for (int jump = 1; jump <= arr[i] && i + jump < n; jump++) {
            ans = Math.min(ans, 1 + solve(i + jump, arr, dp));
        }
        return dp[i] = ans;
    }
    public int minJumps(int[] arr) {
        int n = arr.length;
        int[] dp = new int[n];
        Arrays.fill(dp, -1);
        int ans = solve(0, arr, dp);
        return ans >= (int)1e9 ? -1 : ans;
    }
}
