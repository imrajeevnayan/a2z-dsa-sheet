class Solution {
    public Node flatten(Node head) {
        if (head == null) return null;

        PriorityQueue<Node> pq =
            new PriorityQueue<>((a, b) -> a.data - b.data);

        Node temp = head;

        while (temp != null) {
            pq.offer(temp);
            temp = temp.next;
        }

        Node dummy = new Node(0);
        Node curr = dummy;

        while (!pq.isEmpty()) {
            Node node = pq.poll();

            curr.bottom = node;
            curr = curr.bottom;

            if (node.bottom != null) {
                pq.offer(node.bottom);
            }

            curr.next = null;
        }

        return dummy.bottom;
    }
}
