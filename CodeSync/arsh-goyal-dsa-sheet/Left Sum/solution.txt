import java.util.*;
import java.io.*;
 
public class Solution {    
 
    public static long leftSum(BinaryTreeNode<Integer> root) {
        if (root == null) return 0;
    
        long sum = 0;
        if (root.left != null) {
            sum += root.left.data;
        }
        sum += leftSum(root.left);
        sum += leftSum(root.right);
 
        return sum;
    }
}
