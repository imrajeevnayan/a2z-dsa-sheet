import java.util.ArrayList;
 
public class Solution {
    
    public static ArrayList<Integer> sumTree(BinaryTreeNode<Integer> root) {
        ArrayList<Integer> result = new ArrayList<>();
        computeSum(root, result);
        return result;
    }
    
    private static int computeSum(BinaryTreeNode<Integer> node, ArrayList<Integer> result) {
        if (node == null) return 0;
        
        int leftSum = computeSum(node.left, result);
        int rightSum = computeSum(node.right, result);
        
        int total = node.data + leftSum + rightSum;
        result.add(total);
        
        return total;
    }
}
