    private int[] getPosition(int num, int n) {

        int row = n - 1 - (num - 1) / n;
        int col = (num - 1) % n;

        if (((n - 1 - row) & 1) == 1) {
            col = n - 1 - col;
        }

        return new int[]{row, col};
    }
