class Solution {

    int minCost(int[] height) {
        int n = height.length;
        int[] dp = new int[n];

        // -1 means not calculated yet
        Arrays.fill(dp, -1);
        return solve(n - 1, height, dp);
    }
    int solve(int i, int[] height, int[] dp) {
        if (i == 0)  return 0;
        
        // Already calculated
        if (dp[i] != -1)  return dp[i];
        

        // Jump from previous stone (i - 1)
        int oneJump = solve(i - 1, height, dp)
                + Math.abs(height[i] - height[i - 1]);

        int twoJump = Integer.MAX_VALUE;

        // Jump from i - 2 if possible
        if (i > 1) {
            twoJump = solve(i - 2, height, dp)
                    + Math.abs(height[i] - height[i - 2]);
        }

        // Minimum cost to reach i
        dp[i] = Math.min(oneJump, twoJump);

        return dp[i];
    }
}
