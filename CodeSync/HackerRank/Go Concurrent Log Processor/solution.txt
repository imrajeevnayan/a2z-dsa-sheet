for i := 0; i < int(logsCount); i++ {
        logsItem := readLine(reader)
        logParts := strings.Fields(logsItem)
        if len(logParts) != 2 {
            continue
        }

        statusCode, err1 := strconv.Atoi(logParts[0])
        responseTime, err2 := strconv.Atoi(logParts[1])
        if err1 != nil || err2 != nil {
            continue
        }

        log := Log{
            StatusCode:   statusCode,
            ResponseTime: responseTime,
        }
        logs = append(logs, log)
    }

    result := ProcessLogs(logs)
    for _, v := range result {
        _, _ = fmt.Fprintf(writer, "%d %d %d\n", v.StatusCode, v.CountOfOccurrences, v.
        AverageResponseTime)
    }

    writer.Flush()
}

func readLine(reader *bufio.Reader) string {
    str, _, err := reader.ReadLine()
    if err == io.EOF {
        return ""
    }

    return strings.TrimRight(string(str), "\r\n")
}

func checkError(err error) {
    if err != nil {
        panic(err)
    }
}
