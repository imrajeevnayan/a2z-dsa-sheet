import java.util.* ;
import java.io.*; 
 
public class Solution {
    public static int convert(int N, String str) {
    int ans = 0;
 
    for (int i = 0; i < N; i++) {
        ans = ans * 2 + (str.charAt(i) - '0');
    }
 
    return ans;
}
 
}
