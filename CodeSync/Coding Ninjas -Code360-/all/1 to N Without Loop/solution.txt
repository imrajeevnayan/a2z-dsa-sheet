public class Solution {
 
    public static int[] printNos(int x) {
        int[] ans = new int[x];
        helper(1, x, ans);
        return ans;
    }
 
    private static void helper(int current, int x, int[] ans) {
        if (current > x) {
            return;
        }
 
        ans[current - 1] = current;
        helper(current + 1, x, ans);
    }
}
