public class Solution {
    public static int[] reverseArray(int n, int[] nums) {
        reverse(nums, 0, n - 1);
        return nums;
    }
    static void reverse(int[] nums, int left, int right) {
        if (left >= right) return;
        // swap
        int temp = nums[left];
        nums[left] = nums[right];
        nums[right] = temp;
 
        reverse(nums, left + 1, right - 1);
    }
}
