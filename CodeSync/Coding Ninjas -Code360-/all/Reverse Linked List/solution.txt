import java.io.*;
import java.util.*;
 
public class Solution {
    public static LinkedListNode<Integer> reverseLinkedList(
            LinkedListNode<Integer> head) {
 
        LinkedListNode<Integer> prev = null, curr = head;
 
        while (curr != null) {
            LinkedListNode<Integer> temp = curr.next;
            curr.next = prev;
            prev = curr;
            curr = temp;
        }
        return prev;
    }
}
