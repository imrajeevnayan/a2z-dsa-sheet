import java.util.* ;
import java.io.*; 
public class Solution {
    public static boolean checkUnique(String s){
        // Write your code here.
        char[] ch = s.toCharArray();
        int[] ans = new int[256];
        for(int i = 0; i < ch.length; i++) {
            ans[ch[i]]++;
            if(ans[ch[i]]>1) return false;
        }
        return true;
    }
}
