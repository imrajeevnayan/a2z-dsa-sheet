import java.util.Arrays;
public class Solution {
   public static int[] waveFormArray(int[] arr, int n) {
        Arrays.sort(arr);
        for (int i = 0; i < n - 1; i += 2) {
            int temp = arr[i];
            arr[i] = arr[i + 1];
            arr[i + 1] = temp;
        }
        return arr;
    }
}
