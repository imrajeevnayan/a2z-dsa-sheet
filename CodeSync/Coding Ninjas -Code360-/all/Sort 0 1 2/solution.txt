import java.util.*;
import java.io.*;
public class Solution {
  public static void sort012(int[] arr) {
        int zero = 0, one = 0, two = 0;
 
        for (int x : arr) {
            if (x == 0) zero++;
            else if (x == 1) one++;
            else two++;
        }
 
        int i = 0;
 
        while (zero-- > 0) arr[i++] = 0;
        while (one-- > 0) arr[i++] = 1;
        while (two-- > 0) arr[i++] = 2;
    }
}
