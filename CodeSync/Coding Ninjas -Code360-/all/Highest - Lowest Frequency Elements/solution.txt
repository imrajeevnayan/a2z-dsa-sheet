import java.util.*;
 
public class Solution {
    public static int[] getFrequencies(int []v) {
        Map<Integer, Integer> map = new HashMap<>();
        for (int n: v) map.put(n, map.getOrDefault(n, 0) + 1);
        int maxElement = -1, minElement = -1;
        int maxFreq = 0, minFreq = Integer.MAX_VALUE;
 
        for (int key : map.keySet()) {
            int freq = map.get(key);
            
            // Update max frequency element
            if (freq > maxFreq || (freq == maxFreq && key < maxElement)) {
                maxFreq = freq;
                maxElement = key;
            }
            // Update min frequency element
            if (freq < minFreq || (freq == minFreq && key < minElement)) {
                minFreq = freq;
                minElement = key;
            }
        }
        return new int[]{maxElement, minElement};
    }
}
