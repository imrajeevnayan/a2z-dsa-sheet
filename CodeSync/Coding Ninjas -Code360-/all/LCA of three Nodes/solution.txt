import java.util.ArrayList;
public class Solution {
    public static ArrayList<Integer> findAnagramsIndices(String str, int n, String ptr, int m) {
        ArrayList<Integer> ans = new ArrayList<>();
        int[] count = new int[256];
        for (char ch : ptr.toCharArray())count[ch]++;
        int[] window = new int[256];
        int left = 0;
        for (int right = 0; right < n; right++) {
            window[str.charAt(right)]++;
            if (right - left + 1 > m) {
                window[str.charAt(left)]--;
                left++;
            }
            if (right - left + 1 == m && same(count, window)) {
                ans.add(left);
            }
        }
        return ans;
    }
 
    private static boolean same(int[] a, int[] b) {
        for (int i = 0; i < 256; i++) {
            if (a[i] != b[i]) {
                return false;
            }
        }
        return true;
    }
