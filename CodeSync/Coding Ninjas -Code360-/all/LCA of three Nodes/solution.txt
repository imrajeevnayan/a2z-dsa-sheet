import java.util.* ;
import java.io.*; 
import java.util.ArrayList;
 
public class Solution 
{
    public static ArrayList<Integer> firstNegativeInteger(
        ArrayList<Integer> arr, int k, int n)
    {
        ArrayList<Integer> ans = new ArrayList<>();
        Queue<Integer> q = new LinkedList<>();
 
        int left = 0;
 
        for (int right = 0; right < n; right++) {
 
            // Negative element ka INDEX store karo
            if (arr.get(right) < 0) q.add(right);
        
            // Window size == k
            if (right - left + 1 == k) {
 
                // Window ke bahar ke negative indices remove karo
                while (!q.isEmpty() && q.peek() < left) {
                    q.poll();
                }
                // First negative
                if (!q.isEmpty()) ans.add(arr.get(q.peek()));
                else  ans.add(0);
            
                // Window slide
                left++;
            }
        }
 
        return ans;
    }
}
