import java.util.*; 
import java.io.*; 
 
public class Solution {
    public static int minSubarraySum(ArrayList<Integer> arr, int n, int k) {
        int left = 0,currSum = 0,ans = Integer.MAX_VALUE;
        for (int right = 0; right < n; right++) {
            currSum += arr.get(right);
            if(right-left+1>k){
                currSum-=arr.get(left);
                left++;
            }
            if(right-left+1==k){
                ans=Math.min(ans,currSum);
            }
           
        }
        return ans;
    }
}
