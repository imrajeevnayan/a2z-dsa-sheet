import java.util.PriorityQueue;
 
public class Solution {
    public static class Kthlargest {
 
        private PriorityQueue<Integer> minHeap;
        private int k;
 
        Kthlargest(int k, int[] arr) {
            this.k = k;
            minHeap = new PriorityQueue<>();
 
            for (int num : arr) {
                minHeap.offer(num);
 
                if (minHeap.size() > k) {
                    minHeap.poll();
                }
            }
        }
 
        int add(int num) {
            minHeap.offer(num);
 
            if (minHeap.size() > k) {
                minHeap.poll();
            }
 
            return minHeap.peek();
