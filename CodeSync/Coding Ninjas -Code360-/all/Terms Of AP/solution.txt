import java.util.*;
import java.io.*;
 
public class Solution {
    public static int[] termsOfAP(int x) {
 
        int[] ans = new int[x];
        int count = 0;
        int num = 5;
 
        while (count < x) {
            if (num % 4 != 0) {
                ans[count] = num;
                count++;
            }
            num += 3;
        }
 
        return ans;
    }
}
