import java.util.*;
 
public class Solution {
    public static long pairsWithGivenSum(int arr[], int sum) {
        Map<Integer, Integer> map = new HashMap<>();
        long count = 0;
        for (int num : arr) {
            int rem = sum - num;
            if (map.containsKey(rem))count += map.get(rem);
            map.put(num, map.getOrDefault(num, 0) + 1);
        }
 
        return count;
    }
}
