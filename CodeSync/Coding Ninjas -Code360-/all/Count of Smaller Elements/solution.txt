import java.util.*;
 
public class Solution {
 
    static class Pair {
        int value;
        int index;
 
        Pair(int value, int index) {
            this.value = value;
            this.index = index;
        }
    }
 
    public static int[] countNumber(int n, int[] arr) {
 
        Pair[] nums = new Pair[n];
 
        for (int i = 0; i < n; i++) {
            nums[i] = new Pair(arr[i], i);
        }
 
        int[] ans = new int[n];
 
        mergeSort(nums, 0, n - 1, ans);
 
        return ans;
    }
