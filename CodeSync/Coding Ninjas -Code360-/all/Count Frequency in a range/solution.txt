public class Solution {
    public static int[] countFrequency(int n, int x, int[] nums) {
        int[] freq = new int[n];
        for (int i = 0; i < n; i++) {
            if (nums[i] >= 1 && nums[i] <= n) {
                freq[nums[i] - 1]++;
            }
        }
        return freq;
    }
}
