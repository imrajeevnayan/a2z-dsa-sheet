import java.util.HashSet;
public class Solution {
 
    public static void intersections(int arr1[], int arr2[]) {
        HashSet<Integer> set = new HashSet<>();
 
        for (int num : arr1) {
            set.add(num);
        }
 
        for (int num : arr2) {
            if (set.contains(num)) {
                System.out.print(num + " ");
                set.remove(num);
 
            }
        }
    }
}
