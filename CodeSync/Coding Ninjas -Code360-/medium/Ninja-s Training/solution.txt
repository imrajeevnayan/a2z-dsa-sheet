public class Solution {
 
    public static long countDistinctWayToClimbStair(int n) {
        long[] dp = new long[n + 1];
 
        // -1 means not calculated yet
        for (int i = 0; i <= n; i++) {
            dp[i] = -1;
        }
 
        return solve(n, dp);
    }
 
    private static long solve(int n, long[] dp) {
        // Base case
        if (n == 0 || n == 1) {
            return 1;
        }
 
        // Already calculated
        if (dp[n] != -1) {
            return dp[n];
        }
 
        long MOD = 1000000007;
 
        // Take 1 step OR take 2 steps
        dp[n] = (solve(n - 1, dp) + solve(n - 2, dp)) % MOD;
 
        return dp[n];
    }
}
