import java.util.*;
public class Solution {  
    public static void intersections(int arr1[], int arr2[]) {
        HashMap<Integer, Integer> map = new HashMap<>();
        for (int num : arr2)map.put(num, map.getOrDefault(num, 0) + 1);
        
        
        // Step 2: arr1 scan karo (taaki output arr1 ke order mein aaye)
        for (int num : arr1) {
            int count = map.getOrDefault(num, 0);
            
            // Agar ye element arr2 mein abhi available hai
            if (count > 0) {
                System.out.print(num + " ");
                
                // Ek baar use kar liya ? count kam karo
                // Taaki same element limit se zyada match na ho
                map.put(num, count - 1);
            }
        }
    }
}
