import java.util.* ;
import java.io.*; 
 
public class Solution {
 
    public static boolean isCircular(Node<Integer> head) {
        if (head == null) {
            return false;
        }
 
        Node<Integer> slow = head;
        Node<Integer> fast = head;
 
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
 
            if (slow == fast) {
                return true;
            }
        }
 
        return false;
    }
}
