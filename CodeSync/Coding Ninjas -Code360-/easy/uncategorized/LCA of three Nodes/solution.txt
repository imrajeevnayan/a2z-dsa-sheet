import java.util.*;
import java.io.*;
 
public class Solution {
    public static int findMinimumCost(String str) {
        // Step 1: Odd length check
        if (str.length() % 2 != 0) return -1;
        int openCount = 0, closeCount = 0;
 
        // Step 2: Remove valid pairs using stack logic
        for (char c : str.toCharArray()) {
            if (c == '{') {
                openCount++;
            } else {
                if (openCount > 0) {
                    openCount--; // Match mil gaya
                } else {
                    closeCount++; // Invalid closing brace
                }
            }
        }
 
        // Step 3: Calculate min swaps
        // openCount: Bache hue unmatched '{'
        // closeCount: Bache hue unmatched '}'
        
        // Half of the unmatched opens need to be swapped with half of the unmatched closes.
        // If counts are odd, we need one extra swap for the middle pair.
        
        return (openCount + 1) / 2 + (closeCount + 1) / 2;
    }
}
