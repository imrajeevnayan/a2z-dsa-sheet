public class Solution {
    public static void selectionSort(int[] arr) {
        
        int n = arr.length;
 
        for (int i = 0; i < n - 1; i++) {
 
            int minIndex = i;
 
            // Find minimum element in remaining array
            for (int j = i + 1; j < n; j++) {
                if (arr[j] < arr[minIndex]) {
                    minIndex = j;
                }
            }
 
            // Swap
            int temp = arr[i];
            arr[i] = arr[minIndex];
            arr[minIndex] = temp;
        }
    }
}
