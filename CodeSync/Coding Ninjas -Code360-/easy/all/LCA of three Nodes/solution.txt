public static int[] firstNegative(int[] arr, int n, int k) {
 
        int[] ans = new int[n - k + 1];
        Queue<Integer> q = new LinkedList<>();
 
        int left = 0;
        int index = 0;
 
        for (int right = 0; right < n; right++) {
 
            // Negative element ka index store karo
            if (arr[right] < 0) {
                q.add(right);
            }
 
            // Window size == k
            if (right - left + 1 == k) {
 
                // Window ke bahar wale indices remove karo
                while (!q.isEmpty() && q.peek() < left) {
                    q.poll();
                }
 
                // First negative
                if (q.isEmpty()) {
                    ans[index++] = 0;
                } else {
                    ans[index++] = arr[q.peek()];
                }
 
                // Slide window
                left++;
            }
        }
 
        return ans;
    }
}
