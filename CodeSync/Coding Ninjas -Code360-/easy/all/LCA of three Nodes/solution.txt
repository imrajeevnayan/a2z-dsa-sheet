public class Solution {
 
    public static LinkedListNode<Integer> sortTwoLists(
            LinkedListNode<Integer> first,
            LinkedListNode<Integer> second) {
 
        if (first == null)
            return second;
 
        if (second == null)
            return first;
 
        if (first.data <= second.data) {
 
            first.next =
                sortTwoLists(first.next, second);
 
            return first;
 
        } else {
 
            second.next =
                sortTwoLists(first, second.next);
 
            return second;
        }
    }
}
