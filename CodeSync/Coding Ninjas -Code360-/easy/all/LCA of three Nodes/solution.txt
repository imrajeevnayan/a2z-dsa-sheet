import java.util.*;
import java.io.*;
 
public class Solution {
 
    public static ArrayList<Integer> findDuplicates(int[] arr, int n) {
 
        HashSet<Integer> seen = new HashSet<>();
        HashSet<Integer> duplicate = new HashSet<>();
 
        for (int num : arr) {
 
            if (!seen.add(num)) {
                duplicate.add(num);
            }
        }
 
        return new ArrayList<>(duplicate);
    }
}
