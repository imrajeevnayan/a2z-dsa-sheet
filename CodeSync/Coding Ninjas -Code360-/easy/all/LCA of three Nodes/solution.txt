            int mid = low + (high - low) / 2;
 
            if (canPlace(stalls, k, mid)) {
                ans = mid;
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        return ans;
    }
    private static boolean canPlace(int[] stalls, int k, int distance) {
 
        int count = 1;
        int lastPlaced = stalls[0];
 
        for (int i = 1; i < stalls.length; i++) {
 
            if (stalls[i] - lastPlaced >= distance) {
                count++;
                lastPlaced = stalls[i];
            }
 
            if (count >= k) {
                return true;
            }
        }
 
        return false;
    }
