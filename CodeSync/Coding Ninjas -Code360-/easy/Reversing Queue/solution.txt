import java.util.*;
 
public class Solution {
    public static void reverse(Queue<Integer> q) {
        // Step 1: Queue ke saare elements stack mein daalo
        Stack<Integer> stack = new Stack<>();
        while (!q.isEmpty()) {
            stack.push(q.poll());
        }
        
        // Step 2: Stack se wapas queue mein daalo
        // Stack LIFO hai toh order reverse ho jayega
        while (!stack.isEmpty()) {
            q.add(stack.pop());
        }
    }
}
