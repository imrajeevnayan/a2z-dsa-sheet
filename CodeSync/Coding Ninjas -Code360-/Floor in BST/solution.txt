public class Solution {
    public static int floorInBST(TreeNode<Integer> root, int X) {
        int floor = -1;
        while (root != null) {
 
            if (root.data == X)  return root.data;
            if (root.data < X) {
                // Current node can be a possible floor
                floor = root.data;
                // Try to find a bigger value <= X
                root = root.right;
 
            } else {
                // root.data > X
                // Need smaller value
                root = root.left;
            }
        }
 
        return floor;
    }
}
