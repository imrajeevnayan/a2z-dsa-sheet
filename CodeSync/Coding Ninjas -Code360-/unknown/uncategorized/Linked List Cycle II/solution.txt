public class Solution {
    public static Node firstNode(Node head) {
        if (head == null || head.next == null) return null;
        Node slow = head,fast = head;
 
        // Detect Cycle
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
            if (slow == fast) {
                Node ptr1 = head;
                Node ptr2 = slow;
                while (ptr1 != ptr2) {
                    ptr1 = ptr1.next;
                    ptr2 = ptr2.next;
                }
                return ptr1;
            }
        }
        return null;
    }
 
}
