import java.util.*;
public class Solution {
    public static int splitArray(int[] arr, int m) {
        int low = 0,high = 0;
        for (int num : arr) {
            low = Math.max(low, num);
            high += num;
        }
        int ans = high;
        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (canSplit(arr, m, mid)) {
                ans = mid;
                high = mid - 1;
            } 
            else {
                low = mid + 1;
            }
        }
        return ans;
    }
