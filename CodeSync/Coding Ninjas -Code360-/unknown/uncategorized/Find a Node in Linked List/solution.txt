public class Solution {
    public static int findNode(Node<Integer> head, int n) {
 
        int index = 0;
        Node<Integer> curr = head;
 
        while (curr != null) {
 
            if (curr.data == n) {
                return index;
            }
            curr = curr.next;
            index++;
        }
        return -1;
    }
}
