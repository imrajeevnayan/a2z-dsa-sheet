public class Solution {
    public static boolean isPalindrome(String str) {
        String s1=str.replaceAll("[^a-zA-Z0-9]","").toLowerCase();
        int left=0,right=s1.length()-1;
        while(left<right){
            if(s1.charAt(left)!=s1.charAt(right))return false;
            left++;
            right--;
        }
        return true;
    }
}
