import java.util.*;
import java.io.*;
public class Solution {
    public static int sqrtN(long N) {
        long left = 1, right = N,ans = 0;
        while (left <= right) {
            long mid = left + (right - left) / 2;
            if (mid * mid <= N) {
                ans = mid;
                left = mid + 1;
            } 
            else {
                right = mid - 1;
            }
        }
        return (int) ans;
    }
}
