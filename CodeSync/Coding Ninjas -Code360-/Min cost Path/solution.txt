public class Solution {
 
    public static int minCostPath(int input[][]) {
 
        int m = input.length;
        int n = input[0].length;
 
        int[][] dp = new int[m][n];
 
        dp[0][0] = input[0][0];
 
        // First column
        for (int i = 1; i < m; i++) {
            dp[i][0] = dp[i - 1][0] + input[i][0];
        }
 
        // First row
        for (int j = 1; j < n; j++) {
            dp[0][j] = dp[0][j - 1] + input[0][j];
        }
 
        // Remaining cells
        for (int i = 1; i < m; i++) {
            for (int j = 1; j < n; j++) {
                dp[i][j] = input[i][j]
                        + Math.min(dp[i - 1][j],
                          Math.min(dp[i][j - 1],
                                   dp[i - 1][j - 1]));
            }
        }
 
        return dp[m - 1][n - 1];
    }
}
