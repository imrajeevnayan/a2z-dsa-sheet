import java.util.*;
 
public class Solution {
 
    public static int maximumChocolates(int r, int c, int[][] grid) {
 
        int[][][] dp = new int[r][c][c];
 
        for (int[][] layer : dp) {
            for (int[] row : layer) {
                Arrays.fill(row, -1);
            }
        }
 
        return solve(0, 0, c - 1, grid, r, c, dp);
    }
 
    private static int solve(int i, int j1, int j2,
                             int[][] grid, int r, int c,
                             int[][][] dp) {
 
        if (j1 < 0 || j1 >= c || j2 < 0 || j2 >= c) {
            return (int)-1e9;
        }
 
        if (i == r - 1) {
            if (j1 == j2) {
                return grid[i][j1];
            }
            return grid[i][j1] + grid[i][j2];
        }
 
        if (dp[i][j1][j2] != -1) {
            return dp[i][j1][j2];
        }
 
        int chocolates;
 
        if (j1 == j2) {
            chocolates = grid[i][j1];
