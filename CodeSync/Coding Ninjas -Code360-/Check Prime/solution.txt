import java.util.Scanner;
 
public class Solution {
 
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
 
        int n = sc.nextInt();
 
        if (n < 2) {
            System.out.println(false);
            return;
        }
 
        boolean isPrime = true;
 
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                isPrime = false;
                break;
            }
        }
 
        System.out.println(isPrime);
    }
}
