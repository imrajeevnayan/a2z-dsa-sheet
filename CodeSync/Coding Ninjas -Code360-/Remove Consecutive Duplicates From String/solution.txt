public class Solution {
    public static String removeConsecutive(StringBuilder str) {
        // Edge case: empty string
        if (str.length() == 0) return "";
        
        StringBuilder res = new StringBuilder();
        
        // Pehla character hamesha add hoga
        res.append(str.charAt(0));
        
        // Har character ko pichle se compare karo
        for (int i = 1; i < str.length(); i++) {
            // Agar current char pichle wale se ALAG hai ? add karo
            // Agar SAME hai ? skip karo (consecutive duplicate)
            if (str.charAt(i) != str.charAt(i - 1)) {
                res.append(str.charAt(i));
            }
        }
        
        return res.toString();
    }
}
