public class Solution {
 
    public static long countDistinctWayToClimbStair(int n) {
        long MOD = 1000000007;
 
        if (n == 0 || n == 1) {
            return 1;
        }
 
        long prev2 = 1; // ways(0)
        long prev1 = 1; // ways(1)
 
        for (int i = 2; i <= n; i++) {
            long curr = (prev1 + prev2) % MOD;
            prev2 = prev1;
            prev1 = curr;
        }
 
        return prev1;
    }
}
