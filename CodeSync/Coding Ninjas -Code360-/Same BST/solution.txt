import java.util.*;
 
public class Solution {
 
    public static Boolean checkBSTs( TreeNode<Integer> root1,TreeNode<Integer> root2) {
        Set<Integer> set1 = new HashSet<>();
        Set<Integer> set2 = new HashSet<>();
 
        store(root1, set1);
        store(root2, set2);
 
        return set1.equals(set2);
    }
    private static void store(TreeNode<Integer> root, Set<Integer> set) {
        if (root == null) return;
        
 
        set.add(root.data);
 
        store(root.left, set);
        store(root.right, set);
    }
}
